15def score_of(a, cv, ct, rg, mi, ut, ch, P):
16 s = P["base"]
17 s = s + P["k_use"] * (ut == 1) * 20
18 s = s + P["k_car3"] * (ct == 3) * 20
19 s = s + P["k_car1"] * (ct == 1) * 20
20 s = s + P["k_reg3"] * (rg == 3) * 20
21 s = s + P["k_age"] * np.maximum(a - P["age_hi"], 0)
22 s = s + P["k_young"] * np.maximum(P["age_lo"] - a, 0)
23 s = s + P["k_mile"] * np.log(np.clip(mi, 1500, None) / 15500) * 20
24 s = s - P["k_val"] * (cv - 21)
25 s = s + P["k_ch"] * (ch == 2)
26 return s

29def true_prem(a, cv, ct, rg, mi, ut, ch, rng, CAR_FREQ, CAR_SEV,
30 REGION_RISK, REGION_SEV, CAT_LOAD):
31 eta = (0.012 * (a - 38)
32 + 0.30 * (ct == 3) + 0.086 * (ct == 1)
33 - 0.10 * (ct == 2) + 0.13 * (rg == 3)
34 + 0.186 * (ut == 1)
35 + 0.42 * np.log(np.clip(mi, 1500, None) / 15500)
36 - 0.16 * np.log(np.clip(cv, 4, None) / 21))
37 freq = 0.075 * np.exp(eta)
38 sev = 12000 * CAR_SEV[ct] * REGION_SEV[rg] * np.exp(0.18 * (cv - 21) / 21)
39 return freq * sev + CAT_LOAD

42def evaluate(P, feats, tp, names):
43 a, cv, ct, rg, mi, ut, ch = feats
44 s = score_of(a, cv, ct, rg, mi, ut, ch, P)
45 thr = float(np.quantile(s, 0.56))
46 acc = s <= thr
47 # 1. 各水平承保率
48 rates = {}
49 for col, arr, n_lv in [("car_type", ct, 4), ("region", rg, 5),
50 ("use_type", ut, 2), ("channel", ch, 3)]:
51 rates[col] = [float(acc[arr == i].mean()) for i in range(n_lv)]
52 ok1 = all(0.05 <= r <= 0.70 for v in rates.values() for r in v)
53 # 2. 梯度
54 rej = ~acc
55 q = np.percentile(tp[rej], [10, 50, 90])
56 amed = float(np.median(tp[acc]))
57 ok2 = q[0] < amed < q[2]
58 # 3. 线上错杀率显著更高
59 med = amed
60 wg = []
61 for i in range(3):
62 m = rej & (ch == i)
63 wg.append(float((tp[m] <= med).mean()) if m.sum() else 0.0)
64 ok3 = wg[2] > wg[0] * 1.4 and wg[2] > wg[1] * 1.4
65 return ok1 and ok2 and ok3, rates, (amed, q), wg, thr
