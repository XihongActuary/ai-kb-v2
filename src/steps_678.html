55def pp_predict(f, X):
56 """
57 纯保费 = 预测频率 × 预测损失。
58
59 f 是 step45_models 产出的 fitted 字典。
60 组合③（损失用常数均值）时 sev_m 为 None，自动退化为「频率 × 平均损失」。
61 注意：损失模型是在 log 尺度上训练的，这里必须 exp 回来。
62 """
63 freq = np.clip(f["freq_m"].predict(X), 1e-6, None)
64 if f["sev_m"] is None:
65 sev = np.asarray(float(f.get("sev_const", 0.0)))
66 else:
67 sev = np.exp(np.clip(f["sev_m"].predict(X), -5.0, 12.0))
68 return freq * sev

72# 外推距离（OOD）：拒保人群离训练中心有多远？
73# ==============================================================================
74def ood_scale(Xa):
75 """
76 返回 (均值, 尺度, 无变异列掩码)。
77
78 ★ 这里有一个非常隐蔽、而且会把结果彻底带偏的坑：
79 本案例里「营运」这个哑变量在【承保人群中恒等于 0】（因为营运车全被拒了）。
80 它的标准差就是 0。
81 如果照教科书写 sd = X.std() + 1e-9，分母就成了 1e-9，
82 拒保人群里只要出现一个营运车，那个人的标准化距离就是 1/1e-9 ≈ 十亿，
83 平均距离会被拉到几千万 —— 一个纯粹由数值除零造成的假信号。
84
85 工程上正确的做法：给「训练集中无变异」的列一个先验尺度。
86 对 0/1 哑变量，p≈0.5 时的标准差是 sqrt(0.5×0.5)=0.5，就用 0.5。
87 """
88 mu = Xa.mean(axis=0)
89 sd = Xa.std(axis=0)
90 ref = float(sd.max()) if sd.max() > 0 else 1.0
91 flat = sd < 1e-6 * ref
92 sd = np.where(flat, 0.5, sd)
93 return mu, sd, flat

96def ood_dist(X, mu, sd):
97 """RMS 标准化距离：一个样本在所有维度上平均偏离训练中心多少个标准差。"""
98 z = (X - mu) / sd
99 return np.sqrt((z ** 2).mean(axis=1))

272 # ---- 业务账 -----------------------------------------------------------
273 print("\n【业务账 · 精算】")
274 # ★ 这里非常容易算错，而且错了以后「净增益」永远是负数，很容易被误读成
275 # 「模型没用」。正确口径是：
276 # 保费 P 里已经含了「预期赔款」这一块（综合成本率 COR 就是赔款占比），
277 # 所以【不能再把真实赔款从毛利里减一遍】—— 那是重复扣减。
278 # 正确算法是走增量：
279 # 救回一单带来的毛利变化 = P − COR × 真实赔款
280 # （P 收进来，按真实成本赔掉，剩下的才是净赚）
281 PREM = 3000.0
282 COR = 0.75 # 综合成本率：预期赔款占保费的 75%
283 m_each = PREM - COR * good_cost # 救回一单的实际毛利 = 3000 − 75%×964 ≈ 1,277
284 print(f" 假设：年保费 {PREM:,.0f} 元")
285 print(f" 错杀人群真实赔付成本 {good_cost:>7,.0f} 元/单")
286 print(f" >>> 口径澄清：综合成本率 COR 已经把「预期赔款」算进去了，")
287 print(f" 所以不能先算 PREM×(1−COR) 再减一次真实赔款 —— 那是扣两遍。")
288 print(f" 正确口径是增量：毛利 = 保费 − COR × 真实赔款")
289 print(f" 救回一单毛利 = {PREM:,.0f} − {COR:.0%} × {good_cost:,.0f}"
290 f" = {m_each:>7,.0f} 元")
291 
292 gross = m_each * tp
293 # 误伤的成本：这个人本来不该收，现在收了。他的毛利贡献
294 # = PREM − COR × 误伤组真实赔款，可能为负。
295 # 注意这里用的是「误伤组真实赔款」而不是「错杀组真实赔款」——
296 # 否则等于把两种人的成本混在一起。
297 fpv_all = rej[flag & ~true_good]._true_premium.mean() if fp > 0 else good_cost
298 m_fp = PREM - COR * fpv_all
299 fp_pnl = m_fp * fp
300 net = gross + fp_pnl
301 print(f"\n 按当前召回 {recall:.0%} 计：")
302 print(f" 救回 {tp:,} 件，年毛利 {gross:>14,.0f} 元")
303 print(f" 误伤 {fp:,} 件，年毛利 {fp_pnl:>14,.0f} 元"
304 f"（单均 {m_fp:,.0f} 元）")
305 print(f" 净增益 {net:>14,.0f} 元")
306 print(f"\n >>> 全额召回（{n_good:,} 件）年毛利可达 "
307 f"{m_each*n_good:,.0f} 元。")
308 print(" 对一家中等规模公司，这大约等于一整支销售队伍一年的成本。")
309 print(f" 而当前召回 {recall:.0%} 就已经拿到 {net:,.0f} 元 ——")
310 print(f" 漏掉的那 {fn:,} 件里还躺着 {(m_each-rej[(~flag)&true_good]._true_premium.mean())*fn:,.0f} 元。")

450def step8_bias(df, scores, key="A"):
451 f = scores[key]["f"]
452 hr("步骤 8 / 8 选择偏差的四种修法：不加权 / 仅IPW / 仅结果模型 / 双重稳健")
453 aud = df[(df.accept == 0) & (df.audit == 1)].copy().reset_index(drop=True)
454 rej = df[df.accept == 0].copy().reset_index(drop=True)
455 target = rej._true_premium.mean()
456 print(f"被拒保人群真实平均纯保费（真值，只有模拟器知道）= {target:,.1f} 元")
457 print(f"审计样本 {len(aud):,} 件，占被拒保件 {len(aud)/len(rej):.1%}")
458 print("这是唯一能「看见」拒保人群真实风险的窗口。")
459 
460 # 结果模型 μ(x) 覆盖【全体】被拒保件 —— 残差校正才是 DR 的关键
461 Xr, _ = build_design(rej, use_channel=f["use_ch"])
462 mu_all = pp_predict(f, Xr)
463 Xa, _ = build_design(aud, use_channel=f["use_ch"])
464 mu_aud = pp_predict(f, Xa)
465 
466 y = aud._true_premium.values
467 w = aud.audit_weight.values.astype(float)
468 
469 naive = y.mean() # ① 不加权
470 ipw = np.average(y, weights=w) # ② 仅 IPW
471 res = mu_all.mean() # ③ 仅结果模型
472 dr = res + np.average((y - mu_aud), weights=w) # ④ 双重稳健
473 
474 print(f"\n {'方法':<30}{'估计值':>11}{'偏差':>11}{'偏差率':>10}")
475 tbl = []
476 for nm, v in [("① 不加权（直接取平均）", naive),
477 ("② 仅 IPW（1/抽样概率）", ipw),
478 ("③ 仅结果模型（模型外推）", res),
479 ("④ 双重稳健 DR（外推+残差）", dr)]:
480 print(f" {nm:<28}{v:>11,.1f}{v-target:>+11,.1f}{(v-target)/target:>10.1%}")
481 tbl.append(dict(method=nm, est=float(v), bias=float(v - target),
482 rel_bias=float((v - target) / target)))
483 
484 print("\n【怎么读这张表】")
485 print(f" ① 不加权 偏差 {(naive-target)/target:+.1%}。审计样本有偏，直接用就偏。")
486 print(f" ② 仅 IPW 偏差 {(ipw-target)/target:+.1%}。修正了「渠道比例」这类已知抽样机制，")
487 print(" 但修不了「配合度」这类不可观测的机制。")
488 print(f" ③ 结果模型 偏差 {(res-target)/target:+.1%}。只要模型本身学对了，权重错了也还行。")
489 print(f" ④ DR 偏差 {(dr-target)/target:+.1%}。模型能解释的被模型拿走，")
490 print(" 剩下的残差才交给权重去修 —— 这才是「双重」。")
491 
492 print("\n【为什么叫「双重」】")
493 print(" 它用了两件东西：① 一个结果模型 μ(x)；② 一组抽样权重 w。")
494 print(" DR = mean_all[ μ(x) ] + Σ wᵢ(yᵢ − μ(xᵢ)) / Σ wᵢ")
495 print(" └── 模型外推到全体 ──┘└── 残差交给权重 ──┘")
496 print(" 括号里是残差 —— 模型能解释掉的，模型自己解释掉；")
497 print(" 解释不掉的（抽样机制带偏的部分），交给权重去修。")
498 print(" 两个机制各管一半，所以叫双重。")
499 print("\n >>> 核心承诺：「模型对」或「权重对」，只要有一个对，结果就无偏。")
500 print(" 这是极强的稳健性保证，也是因果推断的主流做法。")
501 
502 print("\n【一个非常容易搞混的细节 —— 本案例真踩过】")
503 print(" 如果只对【审计样本】算 DR，那么 yᵢ − μ(xᵢ) 在求和时被 μ 抵消，")
504 print(" 残差项恒等于 0，DR 就退化成了 IPW —— 双重稳健就白叫了。")
505 print(" 所以第一项必须外推到【全体】被拒保件，这才是 DR 的灵魂。")
506 print(" 代码对照：")
507 print(" ✗ dr = np.average(mu + (y - mu), weights=w) # 残差被抵消")
508 print(" ✓ dr = mu_all.mean() + np.average(y - mu_aud, weights=w)")
509 
510 print("\n【本案例的诚实之处 —— 一个必须说清的概念边界】")
511 print(" 模拟器里 p_audit 是我们自己在抽样时定的，所以「已知」；")
512 print(" 而「配合度」这个不可观测因素，也是模拟器显式塞进去的。")
513 print(" 于是出现一个教学上的矛盾：机制在模拟器里已知，")
514 print(" IPW 却修正不了它 —— 这恰恰是对的。")
515 print(" >>> IPW 只能修正【你能写下来】的抽样机制。")
516 print(" 现实中「谁愿意配合审计」往往没有记录，权重只能估。")
517 print(" DR 之所以主流，正是因为它给了你一条「权重估错了也还能救」的退路。")
518 
519 pd.DataFrame(tbl).to_csv(os.path.join(OUT, "bias_" + key + ".csv"),
520 index=False, encoding="utf-8-sig")
521 return dict(naive=naive, ipw=ipw, res=res, dr=dr, target=target,
522 table=tbl)
