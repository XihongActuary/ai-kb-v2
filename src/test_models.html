50# 1.2 对数链接会把系数「压缩」——真值是线性关系，log 链接下系数必然更小。
51# 这不是 bug，是 GLM 的固有性质：η = log μ 是乘法关系，系数是相对风险。
52# 所以正确的验证是「符号 + 绝对值排序」，而不是「数值等于线性系数」。
53# 教学要点：真值 (2, -1, 0.5) → 拟合 (0.19, -0.09, 0.055)，整体缩了约 10 倍。
54# 但 |2| > |1| > |0.5| 的相对关系必须保住，这才是「模型学到了东西」的证据。
55m2 = PoissonRegressor(alpha=1e-6, max_iter=200).fit(X, y)
56c2 = m2.coef_[1:]
57a2 = np.abs(c2)
58check("对数链接下系数被压缩，但符号与绝对值排序保持",
59 c2[0] > 0 and c2[1] < 0 and c2[2] > 0 and a2[0] > a2[1] > a2[2],
60 f"coef={np.round(c2, 4)} 真值(2,-1,0.5) → |拟合| 排序 {a2[0]:.3f}>{a2[1]:.3f}>{a2[2]:.3f}")

67check("泊松数据下系数回收准确 (0.9, -0.5)",

186# 4.4 训练集拟合：这个测试数据是「指数 × 泊松噪声」，噪声极大
187# （1200 个样本、均值约 10 的纯计数，deviance 天然就在 17 附近）。
188# 关键教训：不能拍一个绝对阈值当标准。
189# 绝对 deviance 取决于数据的噪声水平，同样的 17.7 在低频数据上很糟、
190# 在高频数据上很正常。正确做法是「相对基线改善了多少」。
191# 这里用相对改善幅度断言，与噪声水平解耦。
192const_dev = poisson_deviance(yg, np.full_like(yg, yg.mean()))
193glm_dev = poisson_deviance(yg, PoissonRegressor(alpha=1e-6, max_iter=300).fit(Xg, yg).predict(Xg))
194imp_const = 1 - dev_gb / const_dev
195imp_glm = 1 - dev_gb / glm_dev
196check("GBDT 相对常数模型至少改善 25%",
197 imp_const > 0.25, f"常数={const_dev:.2f} → GBDT={dev_gb:.2f}，改善 {imp_const*100:.1f}%")
198check("GBDT 优于对数线性 GLM（够得到 x1² 项）",
199 imp_glm > 0.0, f"GLM={glm_dev:.2f} → GBDT={dev_gb:.2f}，改善 {imp_glm*100:.1f}%")

201# 4.4b 真正的对照实验：给一个 GLM 结构性抓不到、GBDT 却能抓的信号。
202# 这里用纯 U 型：μ = exp(a + 1.2*(x0²-1))，对数线性模型无论怎么调系数都拟合不了。
203# 独立 RNG，保证可复现，不受上面随机数消耗顺序影响。
204# 注意 x0 必须有界：若用标准正态，尾部 x0² 可到 25，exp(2+1.2*24) 爆到 1e16，
205# 泊松采样直接溢出。教学上这本身是个好例子——数据工程先于模型选择。
206R2 = np.random.default_rng(7)
207_x0 = R2.uniform(-2, 2, 2000)
208Xc = np.column_stack([_x0, R2.normal(size=2000), R2.normal(size=2000)])
209yc = R2.poisson(np.exp(2.0 + 1.2 * (_x0 ** 2 - 1.0))).astype(float)
210dev_const_c = poisson_deviance(yc, np.full_like(yc, yc.mean()))
211dev_glm_c = poisson_deviance(yc, PoissonRegressor(alpha=1e-8, max_iter=400).fit(Xc, yc).predict(Xc))
212gb_c = GBDT(M=80, learning_rate=0.10, max_depth=4, verbose=False).fit(Xc, yc)
213dev_gb_c = poisson_deviance(yc, gb_c.predict(Xc))
214check("U型信号：GLM 结构性失效（相对常数几无改善）",
215 (1 - dev_glm_c / dev_const_c) < 0.05,
216 f"常数={dev_const_c:.1f} → GLM={dev_glm_c:.1f}，仅改善 {(1-dev_glm_c/dev_const_c)*100:.2f}%")
217check("U型信号：GBDT 大幅胜出（至少比 GLM 好 20%）",
218 dev_gb_c < 0.80 * dev_glm_c,
219 f"GLM={dev_glm_c:.1f} → GBDT={dev_gb_c:.1f}，改善 {(1-dev_gb_c/dev_glm_c)*100:.1f}%")
220check("U型信号：x0 是唯一真信号，重要度应居首",
221 np.argmax(gb_c.importances_) == 0, f"imp={np.round(gb_c.importances_, 3)}")

223# 4.5 极端值鲁棒性：注入巨灾，模型不得发散。
224# 注意：这里验的是「不发散」，不是「拟合得好」。
225# 重尾数据本来就拟不准 —— 实测 5 个 300万级样本吃掉 74.8% 的 deviance，
226# 模型对它们只预测到 12 万（差 12 倍）。这不是 bug，是本质困难，

228y_out = yg.copy()

230gb_o = GBDT(M=40, learning_rate=0.12, verbose=False).fit(Xg, y_out)
231p_o = gb_o.predict(Xg)
232dev_o = poisson_deviance(y_out, p_o)
233check("注入 300 万级巨灾后不发散（F 不爆）",
234 np.isfinite(dev_o) and np.all(np.isfinite(gb_o.F_)) and gb_o.F_.max() < 25,
235 f"deviance={dev_o:.1f} F_max={gb_o.F_.max():.2f}")
236check("巨灾样本预测有限且非负",

250# 5.2 AUC 已知答案
251# 完全可分 → 1.0；完全反向 → 0.0；全同分 → 0.5
252check("AUC 完美可分 = 1", abs(auc_score([0, 0, 1, 1], [1, 2, 3, 4]) - 1) < 1e-9)
253check("AUC 完全反向 = 0", abs(auc_score([0, 0, 1, 1], [4, 3, 2, 1])) < 1e-9)
254check("AUC 全同分 = 0.5", abs(auc_score([0, 1, 0, 1], [5, 5, 5, 5]) - 0.5) < 1e-9)
255# 并列值处理：y=[0,0,1,1]，score=[1,1,2,2]。
256# 正确答案是 1.0 而不是 0.5 —— 因为没有任何一对「正负同分」。
257# 4 个(正,负)配对中正样本分数全部更高，所以 AUC=1.0。
258# 教学要点：初学者常误以为「有并列就该打折」。真正的并列案例见下一条。
259a_tie = auc_score([0, 0, 1, 1], [1, 1, 2, 2])
260check("AUC 并列值（组内同分、组间分离）= 1", abs(a_tie - 1.0) < 1e-9, f"{a_tie:.4f}")
261# 真正的并列：正负样本分数完全一样 → 4 对配对全部同分 → 0.5*4/4 = 0.5
262a_half = auc_score([0, 1, 0, 1], [1, 1, 2, 2])
263check("AUC 正负真同分（分数打平）= 0.5", abs(a_half - 0.5) < 1e-9, f"{a_half:.4f}")
264# 暴力枚举交叉验证：秩和法必须与 O(n_pos*n_neg) 枚举完全一致
265_t, _l = [1, 1, 2, 2], [0, 0, 1, 1]
266_bc = _bn = 0
267for _i, _lab in enumerate(_l):
268 if _lab == 1:
269 for _j, _lj in enumerate(_l):
270 if _lj == 0:
271 _bn += 1
272 _bc += 1.0 if _t[_i] > _t[_j] else 0.5 if _t[_i] == _t[_j] else 0.0
273check("AUC 秩和法 == 暴力枚举法", abs(a_tie - _bc / _bn) < 1e-12,
274 f"秩和={a_tie:.4f} 枚举={_bc/_bn:.4f}")
