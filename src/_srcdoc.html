66def _hl(line, state, js=False):
67 """极简高亮：注释 / 字符串 / 关键字 / 数字 / 函数名。
68 state 用于跨行跟踪三引号字符串；js=True 时按 JS 的 // 注释与关键字走。"""
69 out = []
70 i, n = 0, len(line)
71 kw = JS_KW if js else KW
72 
73 if state["doc"]:
74 end = line.find(state["doc"])
75 if end < 0:
76 return f'<span class="ds">{_esc(line)}</span>'
77 out.append(f'<span class="ds">{_esc(line[:end + len(state["doc"])])}</span>')
78 i = end + len(state["doc"])
79 state["doc"] = None
80 
81 while i < n:
82 c = line[i]
83 if c == "#" and not js:
84 out.append(f'<span class="cm">{_esc(line[i:])}</span>')
85 break
86 if js and c == "/" and line[i:i + 2] == "//":
87 out.append(f'<span class="cm">{_esc(line[i:])}</span>')
88 break
89 if js and c == "/" and line[i:i + 2] == "/*":
90 end = line.find("*/", i + 2)
91 if end >= 0: # 单行块注释 /* … */
92 out.append(f'<span class="cm">{_esc(line[i:end + 2])}</span>')
93 i = end + 2
94 continue
95 out.append(f'<span class="cm">{_esc(line[i:])}</span>')
96 state["doc"] = "*/"
97 break
98 if c in "\"'":
99 tri = line[i:i + 3]
100 if tri in ('"""', "'''"):
101 end = line.find(tri, i + 3)
102 if end < 0:
103 out.append(f'<span class="ds">{_esc(line[i:])}</span>')
104 state["doc"] = tri
105 break
106 out.append(f'<span class="ds">{_esc(line[i:end + 3])}</span>')
107 i = end + 3
108 continue
109 m = re.match(r'(?:[rbfu]{0,2})(?:"(?:[^"\\]|\\.)*"|\'(?:[^\'\\]|\\.)*\')', line[i:])
110 if m:
111 out.append(f'<span class="st">{_esc(m.group(0))}</span>')
112 i += m.end()
113 continue
114 out.append(_esc(c))
115 i += 1
116 continue
117 m = re.match(r"\d[\d_.]*", line[i:])
118 if m and not line[i - 1:i].isalnum():
119 out.append(f'<span class="nm">{m.group(0)}</span>')
120 i += m.end()
121 continue
122 m = re.match(r"(?:def|class)\s+([A-Za-z_]\w*)", line[i:])
123 if m and not js:
124 out.append(f'<span class="kw">{m.group(0)[:m.group(0).index(m.group(1))]}'
125 f'</span><span class="fn">{m.group(1)}</span>')
126 i += m.end()
127 continue
128 m = re.match(r"(?:function)\s+([A-Za-z_]\w*)", line[i:])
129 if m and js:
130 out.append(f'<span class="kw">function </span>'
131 f'<span class="fn">{m.group(1)}</span>')
132 i += m.end()
133 continue
134 m = re.match(r"[A-Za-z_$]\w*", line[i:])
135 if m and (i == 0 or not (line[i - 1].isalnum() or line[i - 1] == "_")):
136 w = m.group(0)
137 cls = "kw" if w in kw else ("bi" if w[0].isupper() else "")
138 out.append(f'<span class="{cls}">{w}</span>' if cls else w)
139 i += m.end()
140 continue
141 out.append(_esc(c))
142 i += 1
143 return "".join(out)

243def build_all(root):
244 cdir = os.path.join(root, CODE_DIR)
245 odir = os.path.join(root, OUT_DIR)
246 if not os.path.isdir(odir):

290def span(root, fname, start, end=None, nth=0):
291 """返回 (起始行, 结束行)，1-based 闭区间。
292
293 start : 子串，第 nth 次出现的那一行作为起点（同名方法用它区分）
294 end : 子串；不传就用「下一个同层或更外层的 def / class」的前一行
295 """
296 ls = _lines(root, fname)
297 hit = [i for i, l in enumerate(ls) if start in l]
298 if not hit:
299 return None
300 s = hit[nth] if nth < len(hit) else hit[-1]
301 if end:
302 j = next((i for i in range(s + 1, len(ls)) if end in ls[i]), len(ls) - 1)
303 return (s + 1, j)
304 
305 # 不传 end 时：结束于「下一个同层或更外层的 def / class」。
306 # 对类里的方法（有缩进）要按缩进比较，否则会一直吞到下一个 class。
307 nxt = (r"^\s*(function |/\* -{3,}|// -{3,})" if _is_js(fname)
308 else r"^\s*(def |class |# =====)")
309 ind = len(ls[s]) - len(ls[s].lstrip())
310 for i in range(s + 1, len(ls)):
311 l = ls[i]
312 if not l.strip():
313 continue
314 cur = len(l) - len(l.lstrip())
315 if cur > ind: # 更深的一层，属于本段
316 continue
317 if re.match(nxt, l):
318 return (s + 1, i)
319 return (s + 1, len(ls))

322def link(root, fname, start, end=None, label="", nth=0):
323 """给 B10 用的：一个指向源码片段的链接（新窗口打开）。"""
324 r = span(root, fname, start, end, nth)
325 if not r:
326 return f'<code>{fname}</code>'
327 a, b = r
328 txt = label or fname
329 ln = f"L{a}" if a == b else f"L{a}–{b}" # 单行时不显示成 L281–281
330 return (f'<a class="codelink" href="src/{fname}.html#L{a}-L{b}" '
331 f'target="_blank" rel="noopener">{txt}'
332 f'<span class="cl-ln">{ln}</span></a>')
333
