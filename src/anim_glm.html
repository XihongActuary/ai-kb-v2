2/* ---------------- 真实数据：Poisson 生成，不是示意图 ---------------- */
3var rng = (function(){ var s = 20261003; return function(){ s=(s*1103515245+12345)>>>0; return s/4294967296; };})();
4var N=46, X=[], Y=[];
5var B0T=-0.55, B1T=-0.028; // 真实规律：年龄越大，出险越少
6for(var i=0;i<N;i++){
7 var age=19+Math.floor(rng()*56); // 19~74
8 var lam=Math.exp(B0T+B1T*(age-40));
9 var y=0, L=Math.exp(-lam); // Knuth 泊松采样
10 var p=1,k=0; do{k++; p*=rng();}while(p>L); y=k-1;
11 X.push(age); Y.push(y);
12}
13/* ---------------- IRLS：真实算法，不是演示假动作 ---------------- */
14function irlsStep(b0,b1){
15 // 一次 Newton–Raphson 更新：猜 mu → 算工作响应 z → 加权最小二乘
16 var Sxx=0,Sxz=0,Sx=0,Sz=0,Sz2=0,n=X.length;
17 for(var i=0;i<n;i++){
18 var eta=b0+b1*(X[i]-40);
19 eta=Math.max(Math.min(eta,4),-4);
20 var mu=Math.exp(eta);
21 var w=mu; // Poisson 的权重就是 mu 本身
22 var z=eta+(Y[i]-mu)/mu; // 工作响应：把「看差距」翻译成数字
23 Sxx+=w*(X[i]-40)*(X[i]-40); Sxz+=w*(X[i]-40)*z;
24 Sx+=w*(X[i]-40); Sz+=w*z; Sz2+=w*z*z;
25 }
26 var Sw=n*Sxx-Sx*Sx; // → 2×2 加权最小二乘的解
27 if(Math.abs(Sw)<1e-10) return [b0,b1];
28 var nb0=(Sxx*Sz-Sx*Sxz)/Sw;
29 var nb1=(n*Sxz-Sx*Sz)/Sw;
30 return [nb0,nb1];
31}
32function deviance(b0,b1){
33 var d=0;
34 for(var i=0;i<X.length;i++){
35 var mu=Math.exp(b0+b1*(X[i]-40)), y=Y[i];
36 if(y>0) d+=2*(y*Math.log(y/mu)-(y-mu)); else d+=2*mu;
37 }
38 return d;
39}
40var steps=[{b0:0,b1:0}]; // 第 0 步：啥都不知道，猜「人人都一样」
41for(var s=0;s<7;s++){
42 var p=steps[steps.length-1];
43 var q=irlsStep(p.b0,p.b1);
44 steps.push({b0:q[0],b1:q[1]});
45}
46steps.forEach(function(st){ st.dev=deviance(st.b0,st.b1); });
