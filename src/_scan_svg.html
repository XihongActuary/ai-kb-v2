7def est_w(s, fs):
8 w = 0.0
9 for ch in s:
10 w += fs * (1.0 if ord(ch) > 0x2000 else 0.55)
11 return w

14def _split_g(body, dx=0.0, dy=0.0):
15 """把 <g transform="translate(dx,dy)"> 里的元素坐标还原成绝对坐标。
16
17 不这么做的话，同一段图元被复制四次（translate 不同），
18 扫描器会看到四个「一模一样的 rect」互相重叠 —— 全是误报。
19 """
20 out = []
21 for m in re.finditer(r'<g transform="translate\(([-\d.]+),\s*([-\d.]+)\)">(.*?)</g>',
22 body, re.S):
23 inner = m.group(3)
24 out.append(_shift(inner, dx + float(m.group(1)), dy + float(m.group(2))))
25 out.append(_split_g(re.sub(r'<g transform="translate[^>]*>.*?</g>', "", inner, flags=re.S),
26 dx, dy))
27 return "\n".join(out)

40for fp in sorted(glob.glob(os.path.join(BASE, "*.html"))):
41 src = open(fp, encoding="utf-8").read()
42 for m in re.finditer(r'<svg viewBox="0 0 ([\d.]+) ([\d.]+)"(.*?)</svg>', src, re.S):
43 W, H, body = float(m.group(1)), float(m.group(2)), m.group(3)
44 name = os.path.basename(fp)
45 # 有 <g transform> 的：只扫还原后的绝对坐标，避免重复计数
46 body += _split_g(body)
47 body = re.sub(r'<g transform="translate[^>]*>.*?</g>', "", body, flags=re.S)
48 rects = []
49 for r in re.finditer(r'<rect x="([\d.]+)" y="([\d.]+)" width="([\d.]+)" height="([\d.]+)"', body):
50 x, y, w, h = tuple(map(float, r.groups()))
51 # 铺满整块画布的是背景板，它「包住」所有元素属于设计，不是缺陷
52 if x <= 0 and y <= 0 and w >= W - 1 and h >= H - 1:
53 continue
54 rects.append((x, y, w, h))
55 for i in range(len(rects)):
56 for j in range(i + 1, len(rects)):
57 ax, ay, aw, ah = rects[i]
58 bx, by, bw, bh = rects[j]
59 ox = min(ax + aw, bx + bw) - max(ax, bx)
60 oy = min(ay + ah, by + bh) - max(ay, by)
61 if ox > 1 and oy > 1:
62 print(f"[OVERLAP] {name} rect{rects[i]} vs {rects[j]} ({ox:.0f}x{oy:.0f})")
63 for t in re.finditer(r'<text x="([\d.]+)" y="([\d.]+)" font-size="([\d.]+)"([^>]*)>(.*?)</text>', body):
64 x, y, fs, attr, s = float(t.group(1)), float(t.group(2)), float(t.group(3)), t.group(4), t.group(5)
65 s = re.sub(r"<[^>]+>", "", s)
66 w = est_w(s, fs)
67 anchor = "middle" if "middle" in attr else ("end" if '"end"' in attr else "start")
68 x0 = x - w / 2 if anchor == "middle" else (x - w if anchor == "end" else x)
69 x1 = x0 + w
70 if x0 < -2 or x1 > W + 2 or y > H + 2:
71 print(f"[OUT] {name} text<{s}> x={x} fs={fs} span=[{x0:.0f},{x1:.0f}] W={W:.0f} H={H:.0f}")
72print("scan done")
