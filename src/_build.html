231def _div_end(body, start):
232 """从 start 处的 <div …> 开始，返回它配对 </div> 结束后的下标。"""
233 depth, j = 1, body.find(">", start) + 1
234 while depth:
235 nd = body.find("<div", j)
236 cd = body.find("</div>", j)
237 if cd < 0:
238 return len(body)
239 if 0 <= nd < cd:
240 depth += 1
241 j = nd + 4
242 else:
243 depth -= 1
244 j = cd + 6
245 return j
