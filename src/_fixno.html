1# -*- coding: utf-8 -*-
2"""把正文里残留的旧「第 NN 页」引用，统一改成 navmap 的新页序。
3
4旧编号：段内编号 ch01=35 … ch13=47；新编号：连续页序 ch01=23 … ch13=35（差 12）
5"""
6import re, os
7from navmap import NAV
8 
9NEW = {n[0]: n[2] for n in NAV if n[1]}
10 
11# 上一版用的是「段内编号」(ch01=35…ch13=47)，现在改回连续页序，统一减 12
12OLD2NEW = {f"{n}": f"{n - 12}" for n in range(35, 48)}
13 
14LOG = []
15 
16# 1) 纯文本：按映射表换（此时还没有任何「正确的 35~47」，不会误伤）
17BARE = re.compile(r'第\s?(\d\d)\s?页')
18 
19def fix_bare(m):
20 new = OLD2NEW.get(m.group(1))
21 if not new or new == m.group(1):
22 return m.group(0)
23 LOG.append((m.group(0), f"第 {new} 页"))
24 return f"第 {new} 页"
25 
26# 2) 带链接的：页号一律以 href 指向的文件为准
27LINK = re.compile(r'(<a href="(ch\d\d)\.html[^"]*">)(第 )?(\d\d)(?=[^<]*</a>)')
28 
29def fix_link(m):
30 new = NEW.get(m.group(2))
31 if not new or m.group(4) == new:
32 return m.group(0)
33 LOG.append((m.group(0), m.group(1) + (m.group(3) or "") + new))
34 return m.group(1) + (m.group(3) or "") + new
35 
36root = os.path.dirname(os.path.abspath(__file__))
37for fn in sorted(os.listdir(root)):
38 if not fn.endswith(".py") or fn.startswith("_fix") or fn == "navmap.py":
39 continue
40 p = os.path.join(root, fn)
41 src = open(p, encoding="utf-8").read()
42 out = LINK.sub(fix_link, BARE.sub(fix_bare, src))
43 if out != src:
44 open(p, "w", encoding="utf-8").write(out)
45 print(f"[改] {fn}")
46 
47print(f"\n本轮 {len(LOG)} 处：")
48for a, b in LOG:
49 print(f" {a} -> {b}")
