9/* ---- 绘制越界检查：伪造 canvas context，抓出画到画布外的图元 ---- */
10let TAG = '?';
11const OUT = [];
12function charW(s, fs) {
13 let w = 0;
14 for (const ch of String(s)) w += fs * (ch.charCodeAt(0) > 0x2000 ? 1.0 : 0.55);
15 return w;
16}
17function makeCtx(w, h) {
18 const st = { font: '12px sans-serif' };
19 const fs = () => parseFloat((st.font.match(/([\d.]+)px/) || [0, 12])[1]) || 12;
20 const note = (k, x, y) => OUT.push(`${TAG} ${k}(${(+x).toFixed(1)},${(+y).toFixed(1)}) 画布 ${w}x${h}`);
21 return new Proxy({}, {
22 get: (t, k) => {
23 if (k === 'canvas') return { width: w, height: h };
24 return (...a) => {
25 /* 注意：fillText 的签名是 (text, x, y)，其余图元是 (x, y, ...) */
26 if (k === 'fillText' || k === 'strokeText') {
27 const x = a[1], y = a[2];
28 if (typeof x === 'number' && typeof y === 'number') {
29 if (y < -2 || y > h + 2) note(k + '「' + a[0] + '」', x, y);
30 else {
31 const x1 = x + charW(a[0], fs());
32 if (x < -2 || x1 > w + 2) note(k + '溢出「' + a[0] + '」', x1, y);
33 }
34 }
35 return undefined;
36 }
37 if (k === 'moveTo' || k === 'lineTo') {
38 const [x, y] = a;
39 if (typeof x === 'number' && typeof y === 'number' && (x < -2 || x > w + 2 || y < -2 || y > h + 2))
40 note(k, x, y);
41 }
42 if (k === 'arc') {
43 const [x, y, r] = a;
44 if (typeof x === 'number' && (x - r < -2 || x + r > w + 2 || y - r < -2 || y + r > h + 2))
45 note('arc r=' + (+r).toFixed(1), x, y);
46 }
47 if (k === 'fillRect' || k === 'strokeRect') {
48 const [x, y, ww, hh] = a;
49 if (typeof x === 'number' && (x < -2 || x + ww > w + 2 || y < -2 || y + hh > h + 2))
50 note(k + ' ' + (+ww).toFixed(0) + 'x' + (+hh).toFixed(0), x, y);
51 }
52 return undefined;
53 };
54 },
55 set: (t, k, v) => { st[k] = v; return true; }
56 });
57}

74// 画布尺寸从 HTML 里读真实值，否则 stub 的固定尺寸会让坐标算错、掩盖布局问题
75const CANVAS = {};
76function harvestCanvas(html) {
77 const re = /<canvas id="(\w+)" width="(\d+)" height="(\d+)"/g;
78 let m;
79 while ((m = re.exec(html))) CANVAS[m[1]] = { width: +m[2], height: +m[3] };
80}
81global.document = {
82 getElementById: (id) => {
83 if (!cache[id]) cache[id] = el(id);
84 if (CANVAS[id]) Object.assign(cache[id], CANVAS[id]); // 每次都覆盖：三个动画共用 id「cv」，缓存会串味
85 return cache[id];
86 },
87 createElement: () => el('new'),
88 querySelectorAll: () => []
89};
90global.setInterval = () => 0;
