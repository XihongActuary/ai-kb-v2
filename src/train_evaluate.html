164def build_design(df, use_channel=False):
165 """
166 把原始特征变成模型能吃的设计矩阵。
167
168 分类变量 one-hot（drop_first=True，避免共线）；
169 连续变量做 log / 中心化 —— 右边偏得很厉害（车价 4~95 万，里程 1800~62000），
170 不变换的话 IRLS 的步长会忽大忽小，系数也难解释。
171
172 ★ 一个非常容易踩、而且踩了不报错的坑：
173 水平集必须【冻结】，不能跟着数据走。
174 本案例里 use_type=1（营运）只出现在被拒保人群，承保人群一个都没有。
175 如果直接对两个 DataFrame 各跑一次 get_dummies(drop_first=True)：
176 承保集只有一个水平 0 → drop 掉 → 产出 0 列
177 拒保集有两个水平 → drop 掉 0 → 产出 1 列
178 训练时 10 列、预测时 11 列，模型不会报错，只会算出一堆胡话。
179 真实项目里这就是「上线后预测全错但没人知道为什么」的经典原因。
180 所以这里显式声明要保留哪些水平，并对齐列顺序。
181 """
182 cols = USE_IN_MODEL + (["channel"] if use_channel else [])
183 parts, names = [], []
184 for c in cols:
185 if c in CAT_COLS:
186 lv = LEVEL_NAMES[c]
187 # 冻结水平集：drop 掉第 0 个水平作为基准，其余全部显式生成
188 d = pd.get_dummies(df[c].astype(int), prefix=c)
189 for i, name in enumerate(lv):
190 if i == 0:
191 continue
192 col = f"{c}_{i}"
193 if col not in d.columns:
194 d[col] = 0.0 # 该水平在此子群里不存在 → 全 0
195 parts.append(d[[col]].astype(float))
196 names.append(name)
197 elif c == "age":
198 parts.append((df.age.astype(float) - 38.0).to_frame())
199 names.append("年龄(相对38)")
200 elif c == "car_value":
201 parts.append(np.log(df.car_value.astype(float)).to_frame())
202 names.append("车价(对数)")
203 else: # mileage
204 parts.append(np.log(df.mileage.astype(float)).to_frame())
205 names.append("年里程(对数)")
206 X = pd.concat(parts, axis=1).values.astype(float)
207 return np.nan_to_num(X, nan=0.0, posinf=0.0, neginf=0.0), names

230# ==============================================================================
231# 步骤 4-5：四个模型依次训练 + 评估
232# ==============================================================================
233def step45_models(acc, tr, te):
234 """
235 承保件一进来就撞上一个墙：直接用纯保费做 Poisson 回归，效果非常差。
236 这不是代码写错了，而是重尾数据的本质困难。所以本步骤分三层：
237
238 第 1 层「撞墙」：先老老实实用纯保费建模型，亲眼看到它打不过常数基线。
239 第 2 层「拆解」：回到精算的老本行 —— 频率 × 损失，分开建模，
240 各自用各自对应的尺子（Poisson deviance / 对数尺度 MSE）。
241 第 3 层「组装」：把两个组件拼成纯保费，改用【业务指标】评价：
242 排序能力（AUC、圈选纯度）+ 校准能力（均值偏差）。
243
244 这三层是精算人上手机器学习最自然的路径：
245 先承认问题，再拆解，最后才换尺子重新评价。
246
247 ★ 关于 Tweedie：Tweedie 回归确实是零膨胀+重尾的标准解法
248 （p∈(1,2) 时连接 Gamma 与 Poisson），本案例没有实现它。
249 取而代之的是上面这条更朴素、也更可控的频率—损失路径 ——
250 它本来就是精算的老本行，而且每个组件都能单独解释、单独监控。
251 凡是能说清楚「为什么这么做」的方法，比能跑通的黑盒更值得先学。
252 """
253 hr("步骤 4-5 / 8 四个模型依次登场：每一步多解决了什么？")
254 
255 for tag, use_ch in [("模型A（不含渠道）", False), ("模型B（含渠道）", True)]:
256 X_all, names = build_design(acc, use_channel=use_ch)
257 y_cnt = acc.claim_cnt.values.astype(float) # 出险次数
258 y_amt = acc.claim_amt.values.astype(float) # 赔付金额
259 
260 Xtr, Xte = X_all[tr], X_all[te]
261 cnt_tr, cnt_te = y_cnt[tr], y_cnt[te]
262 pp_tr, pp_te = y_amt[tr], y_amt[te]
263 # 「好司机」标签：真实期望纯保费是否低于承保人群中位数
264 thr_good = float(np.median(acc._true_premium))
265 good_te = (acc._true_premium.values[te] <= thr_good).astype(int)
266 
267 print(f"\n{'='*76}")
268 print(f" {tag} 特征数 = {X_all.shape[1]}")
269 print(f"{'='*76}")
270 
271 # ======================================================================
272 # 第1 层：直接对「纯保费」建模 —— 撞墙，而且撞得非常有教育意义
273 # ======================================================================
274 sub("第 1 层 · 直接对「纯保费」建模（先撞一次墙）")
275 print(" 直觉：既然目标是纯保费，那就直接把它当因变量回归。")
276 print(" 先看数据长什么样：")
277 q = np.quantile(pp_tr, [0.5, 0.9, 0.99, 0.999])
278 print(f" 中位数{q[0]:>10,.0f} 元")
279 print(f" 90分位{q[1]:>10,.0f} 元")
280 print(f" 99分位{q[2]:>10,.0f} 元")
281 print(f" 99.9分位 {q[3]:>10,.0f} 元")

284 # 尾部集中度：这是后面所有麻烦的根源
285 amt = pp_tr[pp_tr > 0]
286 top1 = np.sort(amt)[::-1][:1].sum() / amt.sum()
287 top10 = np.sort(amt)[::-1][:10].sum() / amt.sum()
288 print(f" 最大一笔赔款就占全部赔款 {top1:.1%}，前 10 笔占 {top10:.1%}")
289 print(" 这就是「零膨胀 + 重尾」。均值被尾部拽走，方差远大于均值。")

291 wall = {}
292 wall["常数基线"] = poisson_deviance(pp_te, np.full(len(te), pp_tr.mean()))
293 glm_pp = PoissonRegressor(alpha=0.05, max_iter=200).fit(Xtr, pp_tr)
294 wall["Poisson GLM(纯保费)"] = poisson_deviance(pp_te, glm_pp.predict(Xte))

296 min_samples_leaf=40, verbose=False).fit(Xtr, pp_tr)
297 wall["GBDT(纯保费)"] = poisson_deviance(pp_te, gb_pp.predict(Xte))

299 print(f"\n {'模型':<24}{'测试集 deviance':>18}{'vs 常数基线':>14}")
300 base = wall["常数基线"]
301 for k in ["常数基线", "Poisson GLM(纯保费)", "GBDT(纯保费)"]:
302 v = wall[k]
303 print(f" {k:<24}{v:>18.1f}{v/base*100:>13.1f}%")

305 # ★ 结论必须由实际数字决定，不能写死。
306 # 换一批数据，结论可能从「打不过常数」变成「略好一点点」，
307 # 那时候真正该说的是「改善被巨灾噪声吃掉了」，而不是「模型没学到东西」。
308 best_pp = min(wall["Poisson GLM(纯保费)"], wall["GBDT(纯保费)"])
309 if best_pp >= base:
310 print(f"\n >>> 撞墙成功：两个模型都打不过「直接用平均值」"
311 f"（最优 {best_pp/base*100:.1f}%）。")
312 else:
313 print(f"\n >>> 注意：模型其实【略微】优于常数基线"
314 f"（最优 {best_pp/base*100:.1f}%），")
315 print(" 但改善幅度只有 "
316 f"{(1-best_pp/base)*100:.1f}%，根本不值一提。")
317 print(" 原因不是模型没学到东西，而是这个指标【被污染了】。")
318 print(" 先别急着下结论 —— 后面会证明：")
319 print(" 【模型基本功没问题，是评价指标被少数巨灾赔款绑架了。】")
320 print(" 这是本章最值钱的一课，也是很多人入门机器学习最大的坑。")
321 print(" 怎么证明？看下一个指标：换成正确的尺子，模型立刻说话。")

324 # 第 2 层：拆开看 —— 频率和损失必须用各自对应的尺子量

328 print(" 频率 = 出险次数（零膨胀、计数型、方差≈均值）→ 用 Poisson deviance")
329 print(" 损失 = 每次赔多少（重尾、连续型）→ 用【对数尺度】的 MSE")

335 min_samples_leaf=40, verbose=False).fit(Xtr, cnt_tr)
336 
337 print(f"\n【频率模型 · Poisson GLM】共{len(cnt_tr):,} 个样本，"
338 f"其中出险 {(cnt_tr>0).sum():,} 个")
339 print(" 系数直接读成相对风险 RR = exp(coef)：")
340 print(" RR = 1.35 → 该因子每 +1，出险次数变成 1.35 倍")
341 s = freq_glm.summary(names)
342 for ln in s.splitlines()[:13]:
343 print(" " + ln)

346 imp = freq_gb.importances_
347 for i in np.argsort(-imp)[:8]:
348 if imp[i] > 0.005:
