2"""
3场景一 · 车险定价：数据生成器
4================================================================
5设计目标：把「被错杀的好司机」这个业务问题，写进数据生成机制本身。
6 机器学习不是万能药，而是一面照出「数据里到底缺了什么」的镜子。
7
8核心机制（这是整个案例的灵魂）
9----------------------------------------------------------------
10真实的保险公司只有「被承保的人」才有出险记录。也就是说：
11
12 训练数据 = 已承保 且 已出险 的人
13 看不见的 = 未承保 的人 —— 无论好坏，全都没有标签
14
15于是产生一个统计学上非常难看的局面：
16 「好司机」如果当年被拒保了，就永远不会出现在你的赔付数据里。
17 模型只见过「当年被放行的人如何出险」，
18 却要你去回答「当年被拒的人会不会出险」——
19 这是在训练数据没覆盖的区域做外推。
20
21我们让「拒保规则」犯错，而且错得合情合理：
22 规则惩罚了 use_type（营运） —— 这条是合理的
23 规则也惩罚了 car_type（SUV/新能源）—— 这条部分合理，但过重了
24 规则最不该惩罚的是 channel（线上） —— 这条与风险无关，
25 它衡量的是「数据取得难度」，不是「危险程度」
26最后一条，就是「错杀」的主要来源。它最隐蔽，因为业务上讲得通：
27 「线上渠道风险信息不全，先拒了再说」—— 听起来很谨慎，其实是懒惰。
28
29我们另外保留一个「审计样本」：
30 现实中，保险公司会通过行业数据交换平台，对拒保件做抽样回访，
31 知道其中一小部分人后来是否真的出险了。
32 但抽样只有约 4.6%，且抽样有偏 —— 线上件更难被抽到。
33 所以审计样本能「看到真相」，却不能直接当训练集用：必须加权。
34 权重 = 1 / P(被抽中)，也就是逆概率加权 IPW。
35 这一条约束，恰好逼出了加权与双重稳健方法。
36
37三种人，看得见的标签
38----------------------------------------------------------------
39 全体 20,000 人 真值我们都知道（模拟器知道，用于教学对账）
40 已承保 约 11,000 人 出险记录可见 ← 可用于训练
41 被拒保 约 9,000 人 出险记录不可见 ← 真正的战场
42 审计样本 拒保中的 4.6% 真值可见但有偏 ← 唯一的「偷看」
43
44教学专用优势
45----------------------------------------------------------------
46真实业务里你永远不知道被拒保人群的真实出险率。
47但模拟器知道。所以本页能诚实地说出「模型错得有多离谱」——
48这是课堂专属的奢侈，请勿在汇报中假装业务也有这个奢侈。
49
50运行： python gen_data.py
51输出： data/drivers.csv

106def make_features(rng, n):
107 """生成 n 个投保人的可观测特征"""
108 # 年龄：偏向 25-55 的在职人群
109 age = np.clip(rng.normal(38, 12, n), 18, 75).round().astype(int)
110 
111 # 车辆价值：对数正态，5 万 - 90 万
112 car_value = np.exp(rng.normal(np.log(21), 0.62, n)).round(1)
113 car_value = np.clip(car_value, 4, 95)
114 
115 car_type = rng.choice(4, n, p=[0.44, 0.33, 0.19, 0.04])
116 region = rng.choice(5, n, p=[0.29, 0.20, 0.24, 0.16, 0.11])
117 use_type = rng.choice(2, n, p=[0.89, 0.11])
118 
119 # 线上渠道用户偏年轻、偏经济车型 —— 这正是「数据不全」的来源

121 channel = np.where(
122 rng.random(n) < np.clip(p_online, 0.05, 0.55),
123 2, # 线上

125 )
126

129 mileage = np.clip(mileage, 1800, 62000).astype(int)

137def true_latent(rng, age, car_value, car_type, region, mileage, use_type, channel):
138 """
139 生成真实风险因子 log_freq 与【真实期望纯保费】。
140
141 这里刻意不含 channel —— 因为渠道本身与「危险程度」无关，
142 它只是「信息取得难度」的代理。记住这一点，后面很关键。
143
144 ★ 一个必须讲清楚的口径问题（曾经在这里翻车）：
145 sev 在下面是以【对数正态的中位数】为锚点算出来的。
146 但中位数 ≠ 期望值。对数正态的期望比中位数大 exp(σ²/2) 倍。
147 如果拿中位数当期望，纯保费的「真值」就会和实际模拟出来的赔付对不上，
148 后面所有「模型预测 vs 真实成本」的对照全部失真。
149 所以这里显式做 median → mean 的修正：
150 E[sev] = exp(log_sev + σ²/2)
151 巨灾是【与特征无关】的，所以它对每个人的期望成本都一样，
152 直接加一个常数 CAT_LOAD，不能漏掉 —— 否则真值会系统性偏低。
153 """
154 # ---------- 出险频率 ----------
155 log_freq = (
156 -2.60# 基准频率 ≈ 7.4%
157 + 0.212 * (car_type == 3) # 性能车
158 + 0.086 * (car_type == 1) # SUV
159 - 0.0135 * (car_value - 21) # 车越贵越少出险（也更难修）
160 + 0.142 * (region == 3) # 西部路况
161 + 0.070 * (region == 2)
162 - 0.0068 * (age - 38) # 经验效应，近似线性
163 + 0.052 * np.maximum(age - 62, 0) # 高龄反弹
164 + 0.116 * (use_type == 1) # 营运显著更危险
165 + 0.33 * np.log(np.clip(mileage, 1500, None) / 15500) # 里程弹性
166 - 0.30 * (car_value > 55) # 高价车多为主动全额索赔
167 )
168 freq = np.exp(log_freq)
169 
170 # ---------- 单次损失（对数正态，以中位数为锚） ----------
171 log_sev = (
172 8.55 # 中位数约 5,150 元；修正后均值约 7,800 元
173 + 0.0136 * car_value
174 + 0.186 * (car_type == 3)
175 + 0.100 * (car_type == 2)
176 + 0.084 * (region == 2)
177 - 0.079 * (region == 3)
178 + 0.30 * np.maximum(age - 60, 0) * 0.04
179 )
180 sev_median = np.exp(log_sev)
181 sev_mean = sev_median * np.exp(0.5 * SEV_SIGMA ** 2) # 中位数 → 均值
182 
183 # ---------- 真实期望纯保费 = 频率 × 损失 + 巨灾期望 ----------
184 true_premium = freq * sev_mean + CAT_LOAD
185 
186 return log_freq, freq, sev_median, true_premium

192def crude_score(age, car_value, car_type, region, mileage, use_type, channel):
193 """
194 核保粗糙分（0-100，越高越容易被拒）。
195 这是一套「讲得通但有偏」的业务规则 —— 现实中的核保系统大多如此。
196
197 规则里埋了三处不同性质的东西，务必分清：
198 use_type —— 合理定价因子
199 car_type —— 部分合理，但惩罚过重
200 channel —— 与风险无关，只反映数据是否齐全 ← 错杀的主因
201
202 ★ 一个必须守住的设计约束（改这组系数时踩过）：
203 承保阈值是「在全体分数上取分位数」得到的，
204 所以任何单一水平都不会被整体性地全部拒掉 ——
205 除非它的规则分高到连最低的那一段都进不去。
206 之前的版本里性能车（car_type=3）被加了 +6.0 分，
207 结果 816 件性能车只放行了 18 件（承保率 2.2%）。
208 后果非常严重：模型只有 18 个样本去学「性能车的风险」，
209 学出来必然是「性能车风险低」（因为这 18 个恰好偏年轻偏便宜），
210 于是模型把所有性能车拒保件都推进「预测纯保费最低」那一档。
211 一上线就是 88% 的误伤率 —— 而且错得很隐蔽：
212 业务只看召回率 34%，会觉得「模型有用」。
213
214 >>> 教训：造「错杀」数据时，错杀必须来自【规则偏差】，
215 而不是来自【某类人全被拒】。前者是本案例要讲的病，
216 后者是造数据时自己挖的坑，两者混在一起会让结论彻底失去说服力。
217
218 所以这里的规则分只做「适度加重」，让被拒人群的风险结构
219 呈现梯度（从 1,167 到 1,450），而不是二元对立。
220 >>> 承保规则不能把某个特征变成「决定性开关」。
221 """
222 s = 26.0
223 # —— 真实风险因子：轻度加权，让被拒人群的风险呈【梯度】——
224 s += 0.18 * (use_type == 1) * 20 # 营运：+3.6（真实更贵）
225 s += 0.16 * (car_type == 3) * 20 # 性能车：+3.2（真实更贵）
226 s += 0.10 * (car_type == 1) * 20 # SUV：+2.0（略重）
227 s += 0.15 * (region == 3) * 20 # 西部：+3.0（真实更贵）
228 s += 0.40 * np.maximum(age - 58, 0) # 高龄：加码
229 s += 0.34 * np.maximum(26 - age, 0) # 低龄：加码
230 s += 0.070 * np.log(np.clip(mileage, 1500, None) / 15500) * 20
231 s -= 0.030 * (car_value - 21) # 车贵则减分
232 # —— 偏见：渠道。与风险无关，只因为「信息不全」而加惩罚 ——
233 # ★ 幅度是扫描出来的（见 _tune.py），不是拍脑袋定的。
234 # +11 → 线上承保率 0%（5193 件全在训练集外），案例直接失效
235 # +4 → 线上承保率 23%、错杀率 52% vs 代理 23%，梯度与偏见都在
236 s += 4.0 * (channel == 2) # 线上渠道：+4 ← 关键偏见
237 return s

243def simulate_claims(rng, freq, sev_median, horizon=1.0):
244 """
245 一年观察期内的出险次数与总损失。
246 出险次数~ Poisson(freq) —— 计数型
247 每次损失 ~ LogNormal(中位数=sev) —— 重尾
248 外加一次「偶发巨灾」 —— 与特征无关的纯噪声
249
250 向量化实现（不用 Python 循环，2 万条数据秒级完成）。
251 """
252 n = len(freq)
253 lam = freq * horizon
254 cnt = rng.poisson(lam)
255 
256 # 逐次出险。关键：损失要按【出险人各自的】中位数抽样，
257 # 因为单次损失和车价、车型、地区有关 —— 这正是「损失模型」能学到的信号。
258 # rng.normal 的 loc 可以是数组，但那样就不能再传 size，否则形状对不上。
259 rep = np.repeat(np.arange(n), cnt) # 每张赔款属于谁
260 mu = np.log(np.clip(sev_median[rep], 1.0, None)) + 0.5 * SEV_SIGMA ** 2
261 losses = np.exp(rng.normal(mu, SEV_SIGMA)) # 长度 = 总出险次数
262 total = np.zeros(n)
263 np.add.at(total, rep, losses) # 摊回每个人
264 
265 # 巨灾：独立伯努利，与特征无关（所以任何模型都预测不了它）
266 cat = rng.random(n) < CAT_PROB
267 total[cat] += rng.normal(CAT_LOG_MEDIAN, CAT_SIGMA, int(cat.sum()))
268 
269 return cnt, total.round(2)

275def main():
276 rng = np.random.default_rng(SEED)
277 
278 age, car_value, car_type, region, mileage, use_type, channel = make_features(rng, N_POP)
279 eta, freq, sev, true_premium = true_latent(
280 rng, age, car_value, car_type, region, mileage, use_type, channel)
281 
282 # 真实出险结果 —— 全部人群都有，这是模拟器的「上帝视角」
283 claim_cnt, claim_amt = simulate_claims(rng, freq, sev)
284 
285 # 核保规则打分与承保决策
286 score = crude_score(age, car_value, car_type, region, mileage, use_type, channel)
287 
288 # 阈值微调，使承保率精确落进目标区间。
289 # 做法：直接在分位数上二分。score 越大越容易被拒，所以
290 # 承保率 = P(score <= thr)，thr 取高分位点。
291 # 旧写法是死循环微调 hi 而 lo 从不动，还白跑 400 次，这里换成真二分。
292 lo, hi = ACCEPT_RATE_TARGET
293 target_acc = (lo + hi) / 2
294 srt = np.sort(score)
295 thr = float(np.quantile(srt, target_acc))
296 accept = score <= thr
297 
298 # ------------------------------------------------------------------
299 # 审计样本：只从拒保件中抽，且有偏
300 #
301 # 教学要点（这个坑非常常见）：
302 # ① 抽样概率 π 是「已知但写在代码里的常数」，不是学出来的。
303 # 这正是能做 IPW 的前提。
304 # ② 但 IPW 只在「每个抽样单元内 π 恒定」时才完美还原边际分布。
305 # 如果我们只用 channel 一个变量描述 π，那么在每个渠道内部
306 # 权重是同一个常数，加权 = 重新配比，逐渠道看必然毫无变化。
307 # 真正修正的是「渠道之间的比例」。
308 # ③ 更狠的一层：现实中抽样还依赖「申请人配合度」—— 这种人更年轻、
309 # 留的信息更全、未出险时更愿意接回访电话。而「配合度」我们看不见、
310 # 也没进模型。这种不可观测的偏差，IPW 修不了。
311 # ④ 修不了不代表没办法：只要结果模型本身学得对，
312 # 配合 IPW 就能得到「双重稳健」的结果。这正是下一课的内容。
313 # ------------------------------------------------------------------
314 rejected_idx = np.where(~accept)[0]
315 
316 # 不可观测的「配合度」：年轻 + 线上件更愿意接回访。
317 # 它同时影响被抽中的概率，且与真实风险轻微相关 —— 制造 IPW 修不掉的偏差。
318 coop = (0.5
319 - 0.006 * (age[rejected_idx] - 38) # 年轻人更配合
320 + 0.10 * (channel[rejected_idx] == 2)) # 线上件流程更顺
321 
322 p_audit = (AUDIT_RATE
323 * np.where(channel[rejected_idx] == 2, 0.60, 1.0) # 已知偏差：渠道
324 * np.clip(coop / 0.5, 0.45, 1.60)) # 未知偏差：配合度
325 p_audit = np.clip(p_audit, 0.005, 0.35)
326 
327 audit_flag = np.zeros(N_POP, dtype=bool)
328 audit_idx = rejected_idx[rng.random(len(rejected_idx)) < p_audit]
329 audit_flag[audit_idx] = True
330 
331 # 逆概率权重：w = 1 / π。
332 # 一个被抽中的线上被拒件，只能代表 1/π 个线上被拒件。
333 # 不加权，模型会以为「易抽到的人」样本更多，从而压低他们的预测。
334 audit_weight = np.zeros(N_POP, dtype=float)
335 for j, i in enumerate(rejected_idx):
336 if audit_flag[i]:
337 audit_weight[i] = 1.0 / p_audit[j]
338 
339 # ------------------------------------------------------------------
340 # 组装数据框
341 # ------------------------------------------------------------------
342 df = pd.DataFrame({
343 "driver_id": np.arange(1, N_POP + 1),
344 "age": age,
345 "car_value": car_value,
346 "car_type": car_type,
347 "region": region,
348 "mileage": mileage,
349 "use_type": use_type,
350 "channel": channel,
351 "crude_score": score.round(2),
352 "accept": accept.astype(int),
353 "audit": audit_flag.astype(int),
354 "audit_weight": audit_weight.round(4),
355 # ---- 只有承保 + 观察期结束，才会入账的字段 ----
356 "claim_cnt": np.where(accept, claim_cnt, -1), # -1 表示「无记录」而非「零出险」
357 "claim_amt": np.where(accept, claim_amt, np.nan),
358 # ---- 模拟器保留的真值，仅供教学对账，现实中拿不到 ----
359 "_eta": eta,
360 "_freq": freq.round(5),
361 "_sev_median": sev.round(1), # 中位数，不是均值，别搞混
362 "_true_premium": true_premium.round(1), # 真实【期望】纯保费
363 "_true_cnt": claim_cnt, # 真实【已实现】出险次数
364 "_true_amt": claim_amt.round(2), # 真实【已实现】赔付（围绕期望波动）
365 })
