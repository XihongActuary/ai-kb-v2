41def _kmap():
42 """知识地图：直接从 NAV 生成。
43
44 为什么不手写——手写会和侧栏各说一套，加一页就要改两处还容易漏。
45 这里读的是同一个 NAV，加页、调顺序，地图自动跟上。
46 """
47 # 先把 NAV 切成「段 → [(子组标题 or None, [(fn, title, no)])]」
48 segs, cur = [], None
49 for item in NAV:
50 if item[1] is None:
51 if item[0].startswith("第"):
52 cur = {"name": item[0], "rows": []}
53 segs.append(cur)
54 elif cur is not None:
55 cur["rows"].append([item[0], []])
56 continue
57 if cur is None or not cur["rows"]:
58 cur["rows"].append([None, []])
59 cur["rows"][-1][1].append(item)
60 
61 out = ['<div class="kmap">']
62 for k, seg in enumerate(segs):
63 head = seg["name"].split(" · ")
64 no = head[0] # 「第 0 段」
65 rest = head[1] if len(head) > 1 else ""
66 name = rest.split("：")[0] # 「知识地图」
67 desc = rest.split("：")[1] if "：" in rest else ""
68 n = sum(len(r[1]) for r in seg["rows"])
69 out.append(f'<div class="km-seg s{k}"><div class="km-hd">'
70 f'<span class="km-no">{no}</span>'
71 f'<span class="km-t">{name}</span>'
72 f'<span class="km-d">{desc} · {n} 页</span></div>'
73 f'<div class="km-bd">')
74 for sub, pages in seg["rows"]:
75 if sub:
76 out.append(f'<div style="margin-top:6px;color:var(--cinnabar);'
77 f'font-size:12.5px">{sub}</div>')
78 line = []
79 for fn, title, no2 in pages:
80 line.append(f'<a href="{fn}.html"><code>{no2}</code> {title}</a>')
81 out.append(" · ".join(line))
82 out.append("</div></div>")
83 out.append("</div>")
84 return "\n".join(out)
