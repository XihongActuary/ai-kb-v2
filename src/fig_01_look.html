120def fig2_channel():
121 fig, axes = plt.subplots(1, 2, figsize=(12.5, 5.8), dpi=150)
122 fig.patch.set_facecolor(PAPER)
123 fig.suptitle("图2 偏见长什么样：同样一张保单，线上渠道被拒得更多、更冤",
124 fontsize=16, color=INK, fontweight="bold", y=0.98)
125 
126 names = ["代理渠道", "直销渠道", "线上渠道"]
127 colors = [JADE, AMBER, CINNABAR]
128 
129 # 左：错杀率
130 ax = axes[0]
131 bg(ax)
132 rej = df[df.accept == 0]
133 med = df[df.accept == 1]._true_premium.median()
134 rates = [len(rej[(rej.channel_nm == n) & (rej._true_premium <= med)]) /
135 len(rej[rej.channel_nm == n]) for n in names]
136 bars = ax.bar(names, rates, color=colors, width=.55, zorder=3)
137 for b, r in zip(bars, rates):
138 ax.text(b.get_x() + b.get_width() / 2, r + .018, f"{r:.1%}",
139 ha="center", fontsize=15, color=INK, fontweight="bold")
140 ax.set_ylim(0, .56)
141 ax.set_ylabel("被错杀率（真实纯保费低于承保中位数）", fontsize=11, color=INK)
142 ax.set_title("错杀率：线上渠道是代理渠道的 2.1 倍", fontsize=13,
143 color=CINNABAR, fontweight="bold", pad=12)
144 ax.grid(axis="y", color=LIGHT, lw=.8, zorder=0)
145 # 对比标注
146 ax.annotate("", xy=(2, rates[2]), xytext=(0, rates[0]),
147 arrowprops=dict(arrowstyle="->", color=CINNABAR, lw=2,
148 connectionstyle="arc3,rad=-.25"))
149 ax.text(1.0, max(rates) * .96, "2.1 倍", fontsize=13, color=CINNABAR,
150 fontweight="bold")
151 
152 # 右：真实风险 vs 规则打分（相关性对比）
153 ax2 = axes[1]
154 bg(ax2)
155 sub = df[df.accept == 0]
156 ax2.scatter(sub.crude_score, sub._true_premium, s=9, alpha=.16,
157 color=CINNABAR, edgecolors="none")
158 on = sub[sub.channel_nm == "线上渠道"]
159 ax2.scatter(on.crude_score, on._true_premium, s=9, alpha=.30,
160 color=AMBER, edgecolors="none")
161 # 真实风险趋势线
162 xs = np.linspace(sub.crude_score.min(), sub.crude_score.max(), 50)
163 coef = np.polyfit(sub.crude_score, np.log(sub._true_premium), 1)
164 ax2.plot(xs, np.exp(np.polyval(coef, xs)), color=JADE, lw=2.4,
165 label="真实风险（对数拟合）")
166 ax2.set_xlabel("核保规则打分（越高越容易被拒）", fontsize=11, color=INK)
167 ax2.set_ylabel("真实纯保费（元）", fontsize=11, color=INK)
168 ax2.set_title("规则分能测出部分风险，但把「线上」当成了危险", fontsize=13,
169 color=INK, fontweight="bold", pad=12)
170 ax2.legend(frameon=False, fontsize=11, loc="upper left")
171 ax2.grid(color=LIGHT, lw=.8, zorder=0)
172 ax2.text(0.97, 0.06, "点越靠右下 = 分数高但实际风险低 = 越可能被错杀",
173 transform=ax2.transAxes, ha="right", fontsize=10, color=INK, alpha=.7)
