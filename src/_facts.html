94def auc(y, s):
95 o = np.argsort(s)
96 rk = np.empty(len(s), float)
97 ss = s[o]
98 i = 0
99 while i < len(ss):
100 j = i
101 while j + 1 < len(ss) and ss[j + 1] == ss[i]:
102 j += 1
103 rk[o[i:j + 1]] = (i + j) / 2.0 + 1
104 i = j + 1
105 n1 = float(y.sum())
106 n0 = float(len(y) - n1)
107 return (rk[y == 1].sum() - n1 * (n1 + 1) / 2) / (n1 * n0)

182# ----------------------------------------------------------------- 外推域分域指标
183# 关键教学点：内插域（承保测试集）的表现 ≠ 外推域（拒保人群）的表现。

185far_mask = np.array([v.startswith("远点") for v in sc["是否远点"].astype(str).values])
