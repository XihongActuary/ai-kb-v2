47def poisson_deviance(y, mu, w=None):
48 """
49 Poisson deviance（泊松偏差）：
50 D = 2 · Σ [ y·log(y/μ) - (y - μ) ]
51 直觉：它衡量「预测的次数分布」与「实际的次数分布」差多少。
52 完美预测时 D = 0。D 越小越好。
53 对 y=0 的项约定 0·log0 = 0，不会产生 NaN。
54
55 为什么不用 MSE？
56 赔付金额重尾，MSE 会被少数大额赔案主导，
57 而那些大额赔案本身波动极大、几乎不可预测。
58 Poisson deviance 相当于在「频率」尺度上打分，稳健得多。
59 """
60 y = np.asarray(y, float)
61 mu = np.clip(np.asarray(mu, float), 1e-9, None)
62 w = np.ones(len(y)) if w is None else np.asarray(w, float)
63 with np.errstate(divide="ignore", invalid="ignore"):
64 term = np.where(y > 0, y * np.log(np.maximum(y, 1e-12) / mu), 0.0) - (y - mu)
65 return float(2 * np.sum(w * term) / np.sum(w))

68def auc_score(y, score):
69 """
70 AUC = 「随机取一个正样本和一个负样本，模型给正样本更高分的概率」
71 等价于 ROC 曲线下面积。0.5 = 随机，1.0 = 完美。
72 用秩和公式计算，复杂度 O(n log n)，不画图也能得。
73 """
74 y = np.asarray(y, int)
75 s = np.asarray(score, float)
76 n_pos, n_neg = (y == 1).sum(), (y == 0).sum()
77 if n_pos == 0 or n_neg == 0:
78 return float("nan")
79 # 秩和公式：把分数升序排名（1..n），正样本秩和减去理论最小值
80 order = np.argsort(s, kind="mergesort")
81 ranks = np.empty(len(s), float)
82 ranks[order] = np.arange(1, len(s) + 1)
83 # 并列值必须取平均秩，否则会高估 AUC。
84 # 例：分数 [1,1,2,2] 对应标签 [0,0,1,1] —— 分数越高越可能是正样本，
85 # 而这里两组完全分离，所以 AUC = 1.0（不是 0.5）。
86 # 如果按朴素秩次（1,2,3,4）算，正样本秩和 7，
87 # (7−3)/4 = 1.0，恰好也对；但只要出现「正样本在前、同分在后」
88 # 这类交叉情形，朴素秩就会算错，必须用平均秩。
89 uniq, inv, cnt = np.unique(s, return_inverse=True, return_counts=True)
90 sums = np.zeros(len(cnt))
91 np.add.at(sums, inv, ranks) # 同一分数的秩求和
92 avg_ranks = (sums / cnt)[inv] # 再除以出现次数 = 平均秩
93 return float((avg_ranks[y == 1].sum() - n_pos * (n_pos + 1) / 2) / (n_pos * n_neg))

96def lift_table(y, score, w=None, n_bins=10):
97 """
98 提升度表（Lift / 增益表）：把人群按预测分从高到低分成十等份，
99 看每一份的实际风险是全体平均的多少倍。
100 这是车险定价最常用的一张表 —— 定价员每天看的就是它。
101 """
102 y = np.asarray(y, float)
103 score = np.asarray(score, float)
104 w = np.ones(len(y)) if w is None else np.asarray(w, float)
105 order = np.argsort(-score)
106 y, w = y[order], w[order]
107 tot_w, tot_sy = w.sum(), (w * y).sum()
108 base = tot_sy / tot_w
109 
110 edges = np.linspace(0, len(y), n_bins + 1).astype(int)
111 rows = []
112 cum_w = cum_sy = 0.0
113 for i in range(n_bins):
114 sl = slice(edges[i], edges[i + 1])
115 bw, bs = w[sl].sum(), (w[sl] * y[sl]).sum()
116 cum_w += bw
117 cum_sy += bs
118 seg_rate = bs / bw if bw else np.nan
119 rows.append({
120 "组": f"第{i+1}组",
121 "人数占比": bw / tot_w,
122 "件数占比": bs / tot_sy if tot_sy else np.nan,
123 "该组风险": seg_rate,
124 "Lift": seg_rate / base if base else np.nan,
125 "累计Lift": (cum_sy / cum_w) / base if base else np.nan,
126 "累计人数占比": cum_w / tot_w,
127 })
128 return base, rows

132# 1. Poisson 回归（对数线性 GLM）
133# ==================================================================
134class PoissonRegressor:
135 """
136 建模赔付金额 y：E[y | x] = exp(b0 + b·x)
137
138 损失：Poisson deviance
139 链接：log
140 求解：IRLS（迭代重加权最小二乘）——
141 每一步都在解一个「带权最小二乘」，所以整个算法的骨架
142 和你在本科解回归时解的那个正规方程，是同一个东西。
143
144 为什么不用最小二乘？
145 赔付金额是「大多数接近 0、少数极大」的重尾分布。
146 最小二乘会被几个大额赔案拽跑，把整体拟合压低。
147 Poisson deviance 的惩罚是「按比例」的，稳健得多。
148
149 为什么纯保费建模首选它？
150 ① 预测值恒为正 —— exp 保证了，绝不会出现「负保费」这种荒谬结果
151 ② 系数可解释：exp(b_j) 就是「该因子每升一档，出险强度变几倍」
152 ③ 频率和损失额两条链能用同一套 GLM 表达
153 """
154 L1 = None
155 
156 def __init__(self, alpha=1e-4, max_iter=80, tol=1e-7):
157 self.alpha = alpha # L2 正则强度（等价于岭回归的 λ）
158 self.max_iter = max_iter
159 self.tol = tol
160 
161 def fit(self, X, y, w=None):
162 X = np.asarray(X, float)
163 y = np.asarray(y, float)
164 n, p = X.shape
165 w = np.ones(n) if w is None else np.asarray(w, float)
166 w = w / w.mean()
167 
168 b = np.zeros(p + 1)
169 for it in range(self.max_iter):
170 eta = np.clip(X @ b[1:] + b[0], -20, 20)
171 mu = np.exp(eta)
172 
173 # 工作响应变量 z = η + (y - μ)/μ
174 # 这是牛顿法的一阶展开：让「线性化后的 y」等于 z
175 z = eta + (y - mu) / np.clip(mu, 1e-9, None)
176 W = w * mu # IRLS 权重
177 
178 Xa = np.hstack([np.ones((n, 1)), X])
179 A = (Xa * W[:, None]).T @ Xa
180 A.flat[:: p + 2] += self.alpha # 只罚系数，不罚截距
181 rhs = (Xa * W[:, None]).T @ z
182 try:
183 new_b = np.linalg.solve(A, rhs)
184 except np.linalg.LinAlgError:
185 new_b = np.linalg.lstsq(A, rhs, rcond=None)[0]
186 
187 delta = np.max(np.abs(new_b - b))
188 b = new_b
189 if delta < self.tol:
190 break
191 
192 self.coef_ = b
193 self.n_iter_ = it + 1
194 return self
195 
196 def predict(self, X):
197 X = np.asarray(X, float)
198 return np.exp(np.clip(X @ self.coef_[1:] + self.coef_[0], -20, 20))
199 
200 def summary(self, names=None):
201 """可读输出：系数 + 相对风险 RR = exp(系数)"""
202 names = names or [f"x{i}" for i in range(len(self.coef_) - 1)]
203 lines = ["因子 系数 RR = exp(系数)",
204 "-" * 46]
205 for nm, c in zip(names, self.coef_[1:]):
206 lines.append(f"{nm:<20}{c:>9.4f}{np.exp(c):>12.4f}")
207 lines.append(f"{'(截距)':<20}{self.coef_[0]:>9.4f}")
208 return "\n".join(lines)

214class LogisticRegression:
215 """
216 P(y=1|x) = sigmoid(b0 + b·x)
217 损失：交叉熵（对数损失）
218 求解：牛顿法 / IRLS
219
220 与 Poisson 的关系（精算上很关键的一条直觉）：
221 Poisson 系数 b =「该因子对出险强度的相对影响」
222 Logistic 系数 b =「该因子对是否出险的对数几率的影响」
223 在「泊松对数线性」设定下两者几乎一致 —— 这就是 GLM 家族的统一性，
224 也是为什么精算师掌握了 GLM，就等于掌握了分类模型的一半。
225 """
226 L1 = None
227 
228 def __init__(self, alpha=1e-4, max_iter=60, tol=1e-7):
229 self.alpha = alpha
230 self.max_iter = max_iter
231 self.tol = tol
232 
233 def fit(self, X, y, w=None):
234 X = np.asarray(X, float)
235 y = np.asarray(y, float)
236 n, p = X.shape
237 w = np.ones(n) if w is None else np.asarray(w, float)
238 w = w / w.mean()
239 
240 b = np.zeros(p + 1)
241 for it in range(self.max_iter):
242 eta = np.clip(X @ b[1:] + b[0], -20, 20)
243 phat = np.clip(1 / (1 + np.exp(-eta)), 1e-9, 1 - 1e-9)
244 
245 Xa = np.hstack([np.ones((n, 1)), X])
246 # 梯度
247 g = np.hstack([[np.sum(w * (y - phat))],
248 (X * w[:, None]).T @ (y - phat)]) - self.alpha * b
249 # Hessian
250 W = w * phat * (1 - phat)
251 H = (Xa * W[:, None]).T @ Xa
252 H.flat[:: p + 2] += self.alpha
253 
254 try:
255 step = np.linalg.solve(H, g)
256 except np.linalg.LinAlgError:
257 step = np.linalg.lstsq(H, g, rcond=None)[0]
258 b = b + step
259 if np.max(np.abs(step)) < self.tol:
260 break
261 
262 self.coef_ = b
263 self.n_iter_ = it + 1
264 return self
265 
266 def predict_proba(self, X):
267 X = np.asarray(X, float)
268 eta = np.clip(X @ self.coef_[1:] + self.coef_[0], -20, 20)
269 return 1 / (1 + np.exp(-eta))
270 
271 def predict(self, X, thr=0.5):
272 return (self.predict_proba(X) > thr).astype(int)

278class DecisionTreeRegressor:
279 """
280 一棵树 = 一连串「如果 A > 阈值走左边，否则走右边」。
281
282 怎么找此刻最该问的那个问题？
283 对每个候选切点算「切成两组后，组内平方和减少了多少」，
284 减少最多的切点就是最佳分裂。
285
286 J = Σ左 (y - ȳ左)² + Σ右 (y - ȳ右)²
287 减少量 = J_parent - (J_left + J_right)
288
289 加速技巧（本文件用的）：
290 朴素做法要对每个特征扫 n 个候选值，复杂度 O(n²d)。
291 工程上用「分箱直方图」：把连续特征离散成 B=24 个箱，
292 一次扫描算出每箱的 (权重和, 加权y和)，之后所有切点用前缀和 O(1) 求。
293 复杂度降到 O(n·d·B)，与样本量近似线性。
294
295 复杂度：训练 O(n·d·B·log n) 推理 O(depth) = O(log n)
296 """
297 L1 = None
298 
299 def __init__(self, max_depth=4, min_samples_leaf=40, n_bins=24):
300 self.max_depth = max_depth
301 self.min_samples_leaf = min_samples_leaf
302 self.n_bins = n_bins
303 
304 def fit(self, X, y, w=None):
305 X = np.asarray(X, float)
306 y = np.asarray(y, float)
307 w = np.ones(len(y)) if w is None else np.asarray(w, float)
308 self.n_features_ = X.shape[1]
309 self.tree_ = self._build(X, y, 0, w)
310 return self
311 
312 def _best_split(self, X, y, w):
313 """在所有特征上找最佳分裂点"""
314 tot_w = w.sum()
315 tot_sy = (w * y).sum()
316 if tot_w <= 0:
317 return None
318 parent = tot_sy ** 2 / tot_w # 加权均值的平方和 = (Σwy)²/Σw
319 
320 best = None
321 for j in range(X.shape[1]):
322 col = X[:, j]
323 lo, hi = col.min(), col.max()
324 if hi - lo < 1e-12:
325 continue
326 b = np.clip(((col - lo) / (hi - lo) * self.n_bins).astype(int),
327 0, self.n_bins - 1)
328 if len(np.unique(b)) < 2:
329 continue
330 bw = np.bincount(b, weights=w, minlength=self.n_bins)
331 bs = np.bincount(b, weights=w * y, minlength=self.n_bins)
332 cw = np.cumsum(bw)
333 cs = np.cumsum(bs)
334 ok = (cw >= self.min_samples_leaf) & (tot_w - cw >= self.min_samples_leaf)
335 if not ok.any():
336 continue
337 score = np.full(self.n_bins, -np.inf)
338 score[ok] = (cs[ok] ** 2 / np.maximum(cw[ok], 1e-12)) + \
339 ((tot_sy - cs[ok]) ** 2 / np.maximum(tot_w - cw[ok], 1e-12))
340 k = int(np.argmax(score))
341 gain = score[k] - parent
342 if gain > 0 and (best is None or gain > best[2]):
343 thr = lo + (hi - lo) * (k + 0.5) / self.n_bins
344 best = (j, thr, gain)
345 return best
346 
347 def _build(self, X, y, depth, w):
348 node = {"n": len(y), "pred": float(np.average(y, weights=w)),
349 "leaf": True}
350 if depth >= self.max_depth or len(y) < 2 * self.min_samples_leaf:
351 return node
352 
353 best = self._best_split(X, y, w)
354 if best is None:
355 return node
356 
357 j, thr, _ = best
358 m = X[:, j] < thr
359 node.update(leaf=False, feature=j, threshold=thr,
360 left=self._build(X[m], y[m], depth + 1, w[m]),
361 right=self._build(X[~m], y[~m], depth + 1, w[~m]))
362 return node
363 
364 def predict(self, X):
365 X = np.asarray(X, float)
366 out = np.empty(len(X))
367 for i, x in enumerate(X):
368 nd = self.tree_
369 while not nd["leaf"]:
370 nd = nd["left"] if x[nd["feature"]] < nd["threshold"] else nd["right"]
371 out[i] = nd["pred"]
372 return out
373

375 return sum(1 for _ in iter_nodes(self.tree_))

377 def export_rules(self, names=None):
378 """导出 if-else 文字规则 —— 精算最需要的可解释形式"""
379 names = names or [f"x{i}" for i in range(self.n_features_)]
380 lines = []
381 def rec(nd, cond):
382 if nd["leaf"]:
383 lines.append(f"{' 且 '.join(cond) if cond else '全体'} → 预测 {nd['pred']:.1f}")
384 return

386 rec(nd["left"], cond + [f"{names[j]} ≤ {thr:.3g}"])
387 rec(nd["right"], cond + [f"{names[j]} > {thr:.3g}"])
388 rec(self.tree_, [])
389 return "\n".join(lines)

393# 4. GBDT（梯度提升决策树 · 牛顿步）
394# ==================================================================
395class GBDT:
396 """
397 从「一棵歪树」到「很多棵歪树」：
398 每一步只用一棵很浅的树，去拟合「上一轮没解释掉的部分」。
399
400 残差在回归里是 y - ŷ；更准确的说法是「负梯度」。
401 对 Poisson deviance，负梯度恰好 = μ - y（代码里会推）。
402
403 为什么它统治了表格数据？
404 ① 树的加法结构天然捕获非线性与特征交互（写规则比写公式容易）
405 ② 顺序拟合：每棵树都很弱（depth=3），天然规避过拟合
406 ③ 特征不用标准化、不用线性假设、量纲无关
407 同数据下精度通常比单棵树高 10~30%
408
409 【本实现最关键的一个细节，也是最容易写错的地方】
410 ------------------------------------------------------------------
411 叶子输出必须直接设为「纯增量」 -Σg/Σh ，而不是「树对目标变量的拟合值」。
412
413 朴素写法是：树去拟合 target = F + g/h，然后把树的预测加回 F。
414 这在第一轮就可能出错 —— 因为 target 里含 F 本身（可能高达 20），
415 树拟合出的值自然也带着这 20，加回 F 就等于每轮把 F 重复累加一遍，
416 指数化之后直接爆到 1e12。
417
418 正确写法（XGBoost 论文 Eq.7）：树只负责「分组」，
419 每组的最优步长由 -Σg/Σh 单独算出，这样加回 F 的量级天然受控。
420
421 复杂度：训练 O(M · n · d · B) M=树数 n=样本 d=特征 B=箱数
422 推理 O(M · log n) 仍然很快，可以实时打分
423 """
424 L1 = None
425

449 def _set_leaf_values(self, tree, X, g, h, w):
450 """
451 牛顿步：为每个叶子算 v = -Σ(w·g) / Σ(w·h)，并夹在 ±step_cap。
452 这是 XGBoost 的核心一行代码。直觉：
453 「这个叶子里，预测整体偏高了多少，就往回退多少」
454 退让幅度用二阶信息 h 加权 —— 越不确定的样本，退让越谨慎。
455
456 实现：先把所有样本按树结构路由到叶子，再按叶子聚合。
457 """
458 # 路由：node_of[i] 指向样本 i 落到的叶子节点对象
459 route = np.empty(len(g), dtype=object)
460 for i, x in enumerate(X):
461 nd = tree.tree_
462 while not nd["leaf"]:
463 nd = nd["left"] if x[nd["feature"]] < nd["threshold"] else nd["right"]
464 route[i] = nd
465 
466 # 按叶子聚合 Σ(w·g) 与 Σ(w·h)
467 groups = {}
468 for i, nd in enumerate(route):
469 key = id(nd)
470 if key not in groups:
471 groups[key] = [nd, 0.0, 0.0]
472 groups[key][1] += w[i] * g[i]
473 groups[key][2] += w[i] * h[i]
474 
475 for nd, sg, sh in groups.values():
476 v = -sg / max(sh, 1e-9)
477 nd["value"] = float(np.clip(v, -self.step_cap, self.step_cap))
478 
479 def fit(self, X, y, w=None):
480 X = np.asarray(X, float)
481 y = np.asarray(y, float)
482 w = np.ones(len(y)) if w is None else np.asarray(w, float)
483 w = w / w.mean()
484 
485 f0 = (float(np.log(np.clip(np.average(y, weights=w), 1e-6, None)))
486 if self.base is None else float(self.base))
487 
488 F = np.full(len(y), f0)
489 self.trees_ = []
490 self.f0_ = f0
491 self.importances_ = np.zeros(X.shape[1])
492 
493 for m in range(self.M):
494 mu = np.exp(np.clip(F, -20, 20))
495 # Poisson deviance 对 log 链接的二阶信息
496 g = mu - y
497 h = np.clip(mu, 1e-6, None)
498 
499 # 树只负责「把样本分组」：按 g/h 的绝对值找最该修的群体
500 target = F + np.clip(g / h, -self.step_cap, self.step_cap)
501 tree = DecisionTreeRegressor(max_depth=self.max_depth,
502 min_samples_leaf=self.min_samples_leaf)
503 tree.fit(X, target, w)
504 self._set_leaf_values(tree, X, g, h, w)
505 self.trees_.append(tree)
506 
507 for nd in iter_nodes(tree.tree_):
508 if not nd["leaf"]:
509 self.importances_[nd["feature"]] += 1
510 
511 F = np.clip(F + self.lr * self._eval(tree, X), -20, 20)
512 
513 if self.verbose and (m % 20 == 19 or m == self.M - 1):
514 dev = poisson_deviance(y, np.exp(np.clip(F, -20, 20)), w)
515 print(f" [{self.tag} 轮次 {m+1:>3}/{self.M}] deviance = {dev:.5f}")
516 
517 self.F_ = F
518 self.mu_ = np.exp(np.clip(F, -20, 20))
519 self.importances_ = self.importances_ / max(self.importances_.sum(), 1)
520 return self
521 
522 def predict(self, X):
523 X = np.asarray(X, float)
524 F = np.full(len(X), self.f0_)
525 for t in self.trees_:
526 F = F + self.lr * self._eval(t, X)
527 return np.exp(np.clip(F, -20, 20))

529 def staged_predict(self, X):
530 """逐轮预测，看精度怎么一轮轮涨上去 —— 教学非常好用"""
531 X = np.asarray(X, float)
532 F = np.full(len(X), self.f0_)
533 out = []

535 F = F + self.lr * self._eval(t, X)
536 if (i + 1) % 10 == 0 or i == 0 or i == len(self.trees_) - 1:
537 out.append((i + 1, np.exp(np.clip(F, -20, 20))))
