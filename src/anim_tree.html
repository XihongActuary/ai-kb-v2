11var MINLEAF=5; // 最小叶子人数
12/* ---------------- 不纯度：基尼系数（分类树标准做法） ---------------- */
13function gini(idxs){
14 if(idxs.length===0) return 0;
15 var n1=0;
16 for(var i=0;i<idxs.length;i++) if(D[idxs[i]].r) n1++;
17 var p=n1/idxs.length;
18 return 2*p*(1-p);
19}
20function pOf(idxs){
21 if(!idxs.length) return 0;
22 var n1=0;
23 for(var i=0;i<idxs.length;i++) if(D[idxs[i]].r) n1++;
24 return n1/idxs.length;
25}
26function splitGain(idx,attr,thr){
27 var L=[],R=[];
28 for(var i=0;i<idx.length;i++){
29 if(D[idx[i]][attr]<=thr) L.push(idx[i]); else R.push(idx[i]);
30 }
31 if(L.length<MINLEAF||R.length<MINLEAF) return null; // 叶子太小不切：5 个人的「规律」是噪声
32 var w=(L.length*gini(L)+R.length*gini(R))/idx.length;
33 return {gain:gini(idx)-w,weighted:w,L:L,R:R,nL:L.length,nR:R.length};
34}

52 var best=null, list=[];

54 var vals=idxs.map(function(i){return D[i][attr];}).sort(function(a,b){return a-b;});
55 for(var j=4;j<vals.length-4;j+=3){

74 var FIN=[b1,r2.best,r3.best].filter(function(x){return !!x;});
