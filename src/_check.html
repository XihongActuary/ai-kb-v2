5检查七类问题：
6 1. HTML 标签闭合（div/table/tr/td/h*/p/a/span/aside/main）

29class Checker(HTMLParser):
30 VOID = {"meta", "link", "img", "br", "hr", "input", "source", "wbr"}
31 def __init__(self):
32 super().__init__()
33 self.stack = []
34 self.ids = set()
35 self.hrefs = []
36 self.srcs = []
37 self.has_viewport = False
38 def handle_starttag(self, tag, attrs):
39 d = dict(attrs)
40 if "id" in d:
41 self.ids.add(d["id"])
42 if tag == "a" and d.get("href", "").startswith("#"):
43 self.hrefs.append(d["href"][1:])
44 if tag in ("img", "iframe") and d.get("src"):
45 self.srcs.append(d["src"])
46 if tag == "meta" and d.get("name") == "viewport":
47 self.has_viewport = True
48 if tag not in self.VOID:
49 self.stack.append(tag)
50 def handle_endtag(self, tag):
51 if tag in self.VOID:
52 return
53 if not self.stack:
54 fails.append(f"多余的 </{tag}>")
55 return
56 if self.stack[-1] == tag:
57 self.stack.pop()
58 else:
59 if tag in self.stack:
60 while self.stack and self.stack[-1] != tag:
61 t = self.stack.pop()
62 fails.append(f"未闭合 <{t}>")
63 if self.stack:
64 self.stack.pop()
65 else:
66 fails.append(f"未 opened 的 </{tag}>")
