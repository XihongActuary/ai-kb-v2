2/* ---------------- 真实数据：U 形规律 + 噪声（GLM 最怕的形状） ---------------- */
3var rng=(function(){var s=920241;return function(){s=(s*1103515245+12345)>>>0;return s/4294967296;};})();
4var N=80, X=[], Y=[];
5function truth(x){ return 1.0+0.55*Math.pow((x-50)/22,2); } // 中间低两头高的 U 形
6for(var i=0;i<N;i++){
7 var x=18+rng()*64; // 年龄 18~82
8 var y=truth(x)+(rng()-0.5)*0.34;
9 X.push(x); Y.push(y);
10}
11/* ---------------- GBDT：真实算法（回归树 + 一阶梯度） ---------------- */
12var LR=0.45, MAXD=2; // 学习率 / 树深
13function meanOf(idxs,res){ var m=0; idxs.forEach(function(i){m+=res[i];}); return m/idxs.length; }
14function sseOf(idxs,res){ var m=meanOf(idxs,res),s=0; idxs.forEach(function(i){s+=Math.pow(res[i]-m,2);}); return s; }
15function fitStump(res){
16 /* 深度 2 的回归树：找两刀，让叶子内的残差平方和最小 */
17 var idx=X.map(function(_,i){return i;});
18 function bestSplit(idxs){
19 var bs=null;
20 for(var j=0;j<idxs.length;j++){
21 for(var k2=j+1;k2<idxs.length;k2++){
22 var thr=(X[idxs[j]]+X[idxs[k2]])/2;
23 var L=[],R=[];
24 idxs.forEach(function(i){ (X[i]<=thr?L:R).push(i); });
25 if(L.length<5||R.length<5) continue;
26 var sse=sseOf(L,res)+sseOf(R,res);
27 if(!bs||sse<bs.sse) bs={thr:thr,sse:sse,L:L,R:R};
28 }
29 }
30 return bs;
31 }
32 /* 关键：每个节点的 sse0 必须是它自己的残差平方和，否则「改善不足则停」的判据恒真 */
33 function mk(idxs){ return {idxs:idxs,sse0:sseOf(idxs,res),leaf:meanOf(idxs,res),thr:undefined}; }
34 var root=mk(idx);
35 function grow(node,depth){
36 if(depth>=MAXD) return;
37 var bs=bestSplit(node.idxs);
38 if(!bs) return;
39 if(bs.sse>=node.sse0*0.98) return; // 这一刀只能削掉不到 2% 的误差，不值得切
40 node.thr=bs.thr;
41 node.l=mk(bs.L); node.r=mk(bs.R);
42 grow(node.l,depth+1); grow(node.r,depth+1);
43 }
44 grow(root,0);
45 return root;
46}
47function stumpPredict(root,x){
48 var n=root;
49 while(n.thr!==undefined){ n = x<=n.thr ? n.l : n.r; }
50 return n.leaf;
51}
52function leafCount(n){ return n.thr===undefined?1:leafCount(n.l)+leafCount(n.r); }
53/* 累计预测 = 前 k 棵树的补丁之和；直接走树结构，对任意年龄都能求值 */
54function Fsum(k,x){ var s=0; for(var m=0;m<k;m++) s+=LR*stumpPredict(trees[m],x); return s; }
55var rounds=[{F:X.map(function(){return 0;})}]; // 第 0 轮：从 0 起步（教学版）
56var trees=[];
57var M=8;
58for(var r=0;r<M;r++){
59 var Fp=rounds[rounds.length-1].F;
60 var res=Y.map(function(y,i){return y-Fp[i];});
61 var tree=fitStump(res);
62 trees.push(tree);
63 var Fn=X.map(function(_,i){return Fp[i]+LR*stumpPredict(tree,X[i]);});
64 rounds.push({F:Fn});
65}
66function mse(F){ var s=0; for(var i=0;i<N;i++) s+=Math.pow(Y[i]-F[i],2); return s/N; }
