1/* 复算三个动画的最终数字，写进 out/anim_facts.json。
2 *
3 * 为什么要有这个脚本：
4 * 动画是纯前端算法，跑在浏览器里，Python 流水线够不着它。
5 * 但讲解里要引用的「收敛到多少、误差降到多少」必须是真跑出来的，

7 * 结果落成 JSON，页面再从 JSON 取值 —— 和全站「数字只有一个来源」是同一条规矩。

11 */

17/* ================= ① Poisson GLM · IRLS ================= */
18(function () {
19 var rng = (function () { var s = 20261003; return function () { s = (s * 1103515245 + 12345) >>> 0; return s / 4294967296; }; })();
20 var N = 46, X = [], Y = [], B0T = -0.55, B1T = -0.028;
21 for (var i = 0; i < N; i++) {
22 var age = 19 + Math.floor(rng() * 56);
23 var lam = Math.exp(B0T + B1T * (age - 40));
24 var y = 0, L = Math.exp(-lam), p = 1, k = 0;
25 do { k++; p *= rng(); } while (p > L);
26 y = k - 1; X.push(age); Y.push(y);
27 }
28 function irlsStep(b0, b1) {
29 var Sxx = 0, Sxz = 0, Sx = 0, Sz = 0, n = X.length;
30 for (var i = 0; i < n; i++) {
31 var eta = b0 + b1 * (X[i] - 40);
32 eta = Math.max(Math.min(eta, 4), -4);
33 var mu = Math.exp(eta), w = mu, z = eta + (Y[i] - mu) / mu;
34 Sxx += w * (X[i] - 40) * (X[i] - 40); Sxz += w * (X[i] - 40) * z;
35 Sx += w * (X[i] - 40); Sz += w * z;
36 }
37 var Sw = n * Sxx - Sx * Sx;
38 if (Math.abs(Sw) < 1e-10) return [b0, b1];
39 return [(Sxx * Sz - Sx * Sxz) / Sw, (n * Sxz - Sx * Sz) / Sw];
40 }
41 function dev(b0, b1) {
42 var d = 0;
43 for (var i = 0; i < N; i++) {
44 var mu = Math.exp(b0 + b1 * (X[i] - 40)), y = Y[i];
45 if (y > 0) d += 2 * (y * Math.log(y / mu) - (y - mu)); else d += 2 * mu;
46 }
47 return d;
48 }
49 var st = [{ b0: 0, b1: 0 }];
50 for (var s2 = 0; s2 < 7; s2++) {
51 var p = st[st.length - 1], q = irlsStep(p.b0, p.b1);
52 st.push({ b0: q[0], b1: q[1] });
53 }
54 st.forEach(function (x) { x.dev = dev(x.b0, x.b1); });
55 var last = st[st.length - 1];
56 R.glm = {
57 n: N, rounds: st.length - 1,
58 steps: st.map(function (x) { return { b0: +x.b0.toFixed(4), b1: +x.b1.toFixed(4), dev: +x.dev.toFixed(2) }; }),
59 b0: +last.b0.toFixed(4), b1: +last.b1.toFixed(4),
60 dev0: +st[0].dev.toFixed(2), dev_end: +last.dev.toFixed(2),
61 rr: +Math.exp(last.b1).toFixed(4), // 每大一岁，出险强度乘几倍
62 drop_pct: +(100 * (1 - Math.exp(last.b1))).toFixed(2),
63 b1_true: B1T, b0_true: B0T,
64 drop_pct_true: +(100 * (1 - Math.exp(B1T))).toFixed(2)
65 };

68/* ================= ② 决策树 · 基尼分裂 ================= */
69(function () {
70 var MINLEAF = 5;
71 var rng = (function () { var s = 77031; return function () { s = (s * 1103515245 + 12345) >>> 0; return s / 4294967296; }; })();
72 var N = 56, D = [];
73 for (var i = 0; i < N; i++) {
74 var age = 19 + rng() * 55, mile = 8000 + rng() * 38000;
75 var pp = 1 / (1 + Math.exp(-(0.115 * (28 - age) + 0.00007 * (mile - 15000) - 1.1)));
76 D.push({ age: age, mile: mile, r: rng() < pp });
77 }
78 function gini(ix) {
79 if (!ix.length) return 0;
80 var n1 = 0;
81 for (var i = 0; i < ix.length; i++) if (D[ix[i]].r) n1++;
82 var p = n1 / ix.length; return 2 * p * (1 - p);
83 }
84 function pR(ix) {
85 var n1 = 0;
86 for (var i = 0; i < ix.length; i++) if (D[ix[i]].r) n1++;
87 return n1 / ix.length;
88 }
89 function splitGain(idx, attr, thr) {
90 var L = [], R = [];
91 for (var i = 0; i < idx.length; i++) (D[idx[i]][attr] <= thr ? L : R).push(idx[i]);
92 if (L.length < MINLEAF || R.length < MINLEAF) return null;
93 var w = (L.length * gini(L) + R.length * gini(R)) / idx.length;
94 return { gain: gini(idx) - w, weighted: w, L: L, R: R, nL: L.length, nR: R.length };
95 }
96 function scan(idxs) {
97 var best = null;
98 ["age", "mile"].forEach(function (attr) {
99 var vals = idxs.map(function (i) { return D[i][attr]; }).sort(function (a, b) { return a - b; });
100 for (var j = 4; j < vals.length - 4; j += 3) {
101 var thr = (vals[j] + vals[j + 1]) / 2, g = splitGain(idxs, attr, thr);
102 if (!g) continue;
103 var cand = { attr: attr, thr: thr, g: g, idxs: idxs };
104 if (!best || g.gain > best.g.gain) best = cand;
105 }
106 });
107 return best;
108 }
109 /* pack：把一刀的完整「收支表」都留下来。
110 * 页面上要手算 Gain：切之前 − 切之后，两个数都必须是真跑出来的，不能靠人凑。 */
111 function pack(x, tag, knife) {
112 if (!x) return null;
113 var ix = x.idxs, L = x.g.L, R = x.g.R;
114 var w = x.g.weighted;
115 return {
116 knife: knife, tag: tag,
117 attr: x.attr, thr: +x.thr.toFixed(1),
118 n_parent: ix.length, p_parent: +pR(ix).toFixed(3),
119 gini_before: +gini(ix).toFixed(4),
120 nL: x.g.nL, pL: +pR(L).toFixed(3), gini_L: +gini(L).toFixed(4),
121 nR: x.g.nR, pR: +pR(R).toFixed(3), gini_R: +gini(R).toFixed(4),
122 gini_after: +w.toFixed(4),
123 gain: +x.g.gain.toFixed(4)
124 };
125 }
126 var root = D.map(function (_, i) { return i; });
127 var b1 = scan(root);
128 var b2 = b1 ? scan(b1.g.L) : null;

130 /* 最终叶子：某一层切不动时，它父节点那一整组就是一片叶子 */
131 var leaves = [];
132 function leafOf(sp, side) {
133 var ix = side === "L" ? sp.g.L : sp.g.R;
134 return { n: ix.length, p: +pR(ix).toFixed(3) };
135 }
136 if (b1) {

138 else leaves.push(leafOf(b1, "L"));
139 if (b3) { leaves.push(leafOf(b3, "L")); leaves.push(leafOf(b3, "R")); }

141 }
142 var purest = leaves.reduce(function (a, x) { return (!a || x.p < a.p) ? x : a; }, null);
143 var dirtiest = leaves.reduce(function (a, x) { return (!a || x.p > a.p) ? x : a; }, null);

145 n: N, min_leaf: MINLEAF,
146 n_risk: D.filter(function (d) { return d.r; }).length,
147 root_p: +pR(root).toFixed(3), root_gini: +gini(root).toFixed(4),
148 s1: pack(b1, "全部", 1), s2: pack(b2, "左组", 2), s3: pack(b3, "右组", 3),
149 leaves: leaves,
150 purest: purest, dirtiest: dirtiest,

152 left_small: b1 ? b1.g.L.length : 0
153 };

156/* ================= ③ GBDT · 逐轮补残差 ================= */
157(function () {
158 var rng = (function () { var s = 920241; return function () { s = (s * 1103515245 + 12345) >>> 0; return s / 4294967296; }; })();
159 var N = 80, X = [], Y = [];
160 function truth(x) { return 1.0 + 0.55 * Math.pow((x - 50) / 22, 2); }
161 for (var i = 0; i < N; i++) { var x = 18 + rng() * 64; X.push(x); Y.push(truth(x) + (rng() - 0.5) * 0.34); }
162 var LR = 0.45, MAXD = 2, M = 8;
163 function meanOf(ix, r) { var m = 0; ix.forEach(function (i) { m += r[i]; }); return m / ix.length; }
164 function sseOf(ix, r) { var m = meanOf(ix, r), s = 0; ix.forEach(function (i) { s += Math.pow(r[i] - m, 2); }); return s; }
165 function fitStump(res) {
166 var idx = X.map(function (_, i) { return i; });
167 function bestSplit(idxs) {
168 var bs = null;
169 for (var j = 0; j < idxs.length; j++) for (var k2 = j + 1; k2 < idxs.length; k2++) {
170 var thr = (X[idxs[j]] + X[idxs[k2]]) / 2, L = [], R = [];
171 idxs.forEach(function (i) { (X[i] <= thr ? L : R).push(i); });
172 if (L.length < 5 || R.length < 5) continue;
173 var sse = sseOf(L, res) + sseOf(R, res);
174 if (!bs || sse < bs.sse) bs = { thr: thr, sse: sse, L: L, R: R };
175 }
176 return bs;
177 }
178 function mk(ix) { return { idxs: ix, sse0: sseOf(ix, res), leaf: meanOf(ix, res), thr: undefined }; }
179 var rt = mk(idx);
180 function grow(node, depth) {
181 if (depth >= MAXD) return;
182 var bs = bestSplit(node.idxs); if (!bs) return;
183 if (bs.sse >= node.sse0 * 0.98) return;
184 node.thr = bs.thr; node.l = mk(bs.L); node.r = mk(bs.R);
185 grow(node.l, depth + 1); grow(node.r, depth + 1);
186 }
187 grow(rt, 0); return rt;
188 }
189 function sp(n, x) { while (n.thr !== undefined) { n = x <= n.thr ? n.l : n.r; } return n.leaf; }
190 function lc(n) { return n.thr === undefined ? 1 : lc(n.l) + lc(n.r); }
191 var trees = [], rounds = [{ F: X.map(function () { return 0; }) }];
192 for (var r = 0; r < M; r++) {
193 var Fp = rounds[rounds.length - 1].F;
194 var res = Y.map(function (y, i) { return y - Fp[i]; });
195 var t = fitStump(res); trees.push(t);
196 rounds.push({ F: X.map(function (_, i) { return Fp[i] + LR * sp(t, X[i]); }) });
197 }
198 function mse(F) { var s = 0; for (var i = 0; i < N; i++) s += Math.pow(Y[i] - F[i], 2); return s / N; }
199 var noise = 0;
200 for (var i = 0; i < N; i++) noise += Math.pow(Y[i] - truth(X[i]), 2);
201 noise /= N;
202 R.gbdt = {
203 n: N, lr: LR, max_depth: MAXD, rounds: M,
204 mse: rounds.map(function (x) { return +mse(x.F).toFixed(4); }),
205 noise_emp: +noise.toFixed(4),
206 noise_theory: +(0.34 * 0.34 / 12).toFixed(4), // 均匀噪声 ±0.17 的理论方差
207 leaf_counts: trees.map(lc)
208 };
209})();
